// 던전 입구 포털 — 소용돌이 + 빛나는 테두리 + 반짝이.
// 텍스처 없이 노이즈를 직접 계산한다. _SwirlTex를 넣으면 노이즈 대신 그 무늬로 소용돌이를 그린다.
// 색은 HDR이라 1을 넘는 부분은 블룸으로 번진다.
Shader "Proto/Portal"
{
    Properties
    {
        [HDR] _ColorDeep ("Deep", Color) = (0.03, 0.0, 0.10, 1)
        [HDR] _ColorMid  ("Swirl Arms", Color) = (0.42, 0.10, 0.95, 1)
        [HDR] _ColorCore ("Core Glow", Color) = (0.90, 0.55, 1.60, 1)
        [HDR] _RimColor  ("Rim", Color) = (1.80, 0.55, 1.60, 1)
        _Intensity  ("Intensity", Float) = 1.0
        _Speed      ("Spin Speed", Float) = 0.55
        _Twist      ("Twist", Float) = 3.2
        _NoiseScale ("Noise Scale", Float) = 2.6
        _Aspect     ("Aspect (w/h)", Float) = 0.83
        _Corner     ("Corner Round", Range(0, 1)) = 0.45
        _EdgeWobble ("Edge Wobble", Float) = 0.07
        _RimWidth   ("Rim Sharpness", Float) = 26
        _Stars      ("Sparkles", Range(0, 1)) = 0.12
        _SwirlTex   ("Swirl Texture (optional)", 2D) = "black" {}
        _UseTex     ("Use Swirl Texture", Range(0, 1)) = 0
    }

    SubShader
    {
        Tags { "RenderType" = "Transparent" "Queue" = "Transparent" "RenderPipeline" = "UniversalPipeline" "IgnoreProjector" = "True" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            Name "Portal"
            Tags { "LightMode" = "UniversalForward" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_SwirlTex); SAMPLER(sampler_SwirlTex);

            CBUFFER_START(UnityPerMaterial)
                float4 _ColorDeep, _ColorMid, _ColorCore, _RimColor;
                float _Intensity, _Speed, _Twist, _NoiseScale, _Aspect, _Corner, _EdgeWobble, _RimWidth, _Stars, _UseTex;
                float4 _SwirlTex_ST;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            Varyings vert(Attributes v)
            {
                Varyings o;
                o.positionCS = TransformObjectToHClip(v.positionOS.xyz);
                o.uv = v.uv;
                return o;
            }

            float hash(float2 p)
            {
                p = frac(p * float2(123.34, 456.21));
                p += dot(p, p + 45.32);
                return frac(p.x * p.y);
            }

            float vnoise(float2 p)
            {
                float2 i = floor(p), f = frac(p);
                float a = hash(i), b = hash(i + float2(1, 0)), c = hash(i + float2(0, 1)), d = hash(i + float2(1, 1));
                float2 u = f * f * (3.0 - 2.0 * f);
                return lerp(lerp(a, b, u.x), lerp(c, d, u.x), u.y);
            }

            float fbm(float2 p)
            {
                float s = 0, a = 0.5;
                [unroll] for (int i = 0; i < 4; i++) { s += a * vnoise(p); p = p * 2.03 + 17.1; a *= 0.5; }
                return s;
            }

            // 둥근 사각형까지의 거리 (안쪽이 음수)
            float sdRoundRect(float2 p, float2 b, float r)
            {
                float2 q = abs(p) - b + r;
                return length(max(q, 0)) + min(max(q.x, q.y), 0) - r;
            }

            float4 frag(Varyings i) : SV_Target
            {
                float t = _Time.y;
                float2 p = i.uv * 2.0 - 1.0;              // -1 ~ 1
                float2 ps = float2(p.x * _Aspect, p.y);    // 실제 비율

                // 모양 — 도리이 기둥 사이를 채우는 둥근 사각형. 가장자리는 불꽃처럼 일렁인다.
                float d = sdRoundRect(ps, float2(_Aspect, 1.0) * 0.9, _Corner * min(_Aspect, 1.0));
                d += (fbm(ps * 4.0 + float2(0, t * 0.9)) - 0.5) * _EdgeWobble;
                float inside = saturate(-d / 0.035);
                float rim = exp(-abs(d) * _RimWidth);

                // 소용돌이 — 가운데로 갈수록 더 세게 비튼다
                float r = length(ps);
                float ang = _Twist / (r + 0.35) + t * _Speed;
                float s, c; sincos(ang, s, c);
                float2 q = float2(c * ps.x - s * ps.y, s * ps.x + c * ps.y);

                float n = fbm(q * _NoiseScale + float2(0, -t * 0.3));
                float n2 = fbm(q * _NoiseScale * 2.3 - t * 0.25);
                if (_UseTex > 0.5)
                {
                    float tex = SAMPLE_TEXTURE2D(_SwirlTex, sampler_SwirlTex, q * 0.5 + 0.5).r;
                    n = lerp(n, tex, 0.85);
                }
                float arms = pow(saturate(n * 1.3), 3.0);   // 팔과 틈의 대비 — 틈은 깊은 어둠으로

                float3 col = lerp(_ColorDeep.rgb, _ColorMid.rgb, arms);
                col += _ColorCore.rgb * exp(-r * r * 5.0) * (0.75 + 0.25 * sin(t * 2.2));   // 숨 쉬는 중심
                col += _ColorCore.rgb * pow(saturate(n2 * 1.3 - 0.35), 3.0) * 0.7;     // 가는 빛줄기

                // 반짝이 — 소용돌이를 따라 돈다
                float2 g = q * 13.0;
                float h = hash(floor(g));
                float star = step(1.0 - _Stars, h) * saturate(1.0 - length(frac(g) - 0.5) * 6.0) * (0.5 + 0.5 * sin(t * 5.0 + h * 40.0));
                col += star * _ColorCore.rgb * 1.6 * inside;

                col = col * inside + _RimColor.rgb * rim * 1.6;
                col *= _Intensity;

                float alpha = saturate(inside * 0.97 + rim * 0.85);
                return float4(col, alpha);
            }
            ENDHLSL
        }
    }
}
