        static readonly Color ColPreviewRing = new Color(0.30f, 0.32f, 0.42f, 1f);
        static readonly Color ColPreviewFill = new Color(0.12f, 0.11f, 0.16f, 1f);
        static readonly Color ColPreviewEdge = new Color(0.28f, 0.27f, 0.36f, 0.8f);

        /// <summary>
        /// 갈래별로 줄지어 배치한다. 위·아래 갈래는 좌우 2칸, 좌·우 갈래는 위아래 4칸.
        /// 가운데 축은 실제 노드 자리로 비워 둔다.
        /// </summary>
        void BuildPreview()
        {
            _preview = UiKit.Rect(_content, "Preview");
            _preview.SetSiblingIndex(0);   // 실제 노드 아래에 깔린다
            var edges = UiKit.Rect(_preview, "Edges");
            var nodes = UiKit.Rect(_preview, "Nodes");

            var count = new Dictionary<string, int>();
            foreach (var d in PreviewData)
            {
                string br = d[0];
                count.TryGetValue(br, out int i);
                count[br] = i + 1;

                Vector2 dir, side; float start; float[] slots;
                switch (br)
                {
                    case "E": dir = Vector2.up; side = Vector2.right; start = 150f; slots = new[] { -150f, 150f }; break;
                    case "H": dir = Vector2.down; side = Vector2.right; start = 150f; slots = new[] { -150f, 150f }; break;
                    case "M": dir = Vector2.right; side = Vector2.up; start = 340f; slots = new[] { -150f, 150f, -300f, 300f }; break;
                    default: dir = Vector2.left; side = Vector2.up; start = 340f; slots = new[] { -150f, 150f, -300f, 300f }; break;
                }
                int row = i / slots.Length;
                float along = start + row * 150f;
                Vector2 pos = dir * along + side * slots[i % slots.Length];
                Vector2 anchor = dir * (along - 150f);   // 앞 줄의 축 지점에서 뻗는다

                var e = UiKit.Image(edges, "Edge", ColPreviewEdge);
                Vector2 dv = pos - anchor;
                UiKit.Place(e.rectTransform, new Vector2(0.5f, 0.5f), (anchor + pos) * 0.5f, new Vector2(dv.magnitude, 4f));
                e.rectTransform.localRotation = Quaternion.Euler(0f, 0f, Mathf.Atan2(dv.y, dv.x) * Mathf.Rad2Deg);

                bool key = d[4] == "1";
                var v = MakeCircle(nodes, "Preview_" + d[1], pos, key ? 86f : 66f);
                v.Preview = true;
                v.Data = new UpgradeNode { Name = d[1], Effect = d[2], MaxLevel = 1, Keystone = key };
                v.Icon.text = d[1].Substring(0, 1);
                SetArt(v, d[3]);
                v.Ring.color = key ? new Color(0.62f, 0.50f, 0.28f, 1f) : ColPreviewRing;
                v.Fill.color = ColPreviewFill;
                v.Art.color = new Color(0.62f, 0.62f, 0.66f, 1f);
                v.Plate.gameObject.SetActive(true);
                v.Level.text = "";
                v.Cost.text = "<color=#8f8aa3>" + d[1] + "</color>";

                var hover = v.Fill.gameObject.AddComponent<TreeNodeHover>();
                var captured = v;
                hover.Enter = () => ShowTip(captured);
                hover.Exit = () => { if (_hover == captured) HideTip(); };
                hover.Click = () => StartCoroutine(Shake(captured.Root));
            }

            // 갈래 축 — 줄 사이를 잇는 흐린 선
            foreach (var kv in count)
            {
                Vector2 dir = kv.Key == "E" ? Vector2.up : kv.Key == "H" ? Vector2.down : kv.Key == "M" ? Vector2.right : Vector2.left;
                int per = (kv.Key == "E" || kv.Key == "H") ? 2 : 4;
                float start = (kv.Key == "E" || kv.Key == "H") ? 150f : 340f;
                float end = start + ((kv.Value - 1) / per - 1) * 150f;
                if (end <= 0f) continue;
                var e = UiKit.Image(edges, "Spine_" + kv.Key, ColPreviewEdge);
                UiKit.Place(e.rectTransform, new Vector2(0.5f, 0.5f), dir * end * 0.5f, new Vector2(end, 4f));
                e.rectTransform.localRotation = Quaternion.Euler(0f, 0f, Mathf.Atan2(dir.y, dir.x) * Mathf.Rad2Deg);
                e.transform.SetAsFirstSibling();
            }

            _preview.gameObject.SetActive(false);
        }

        void TogglePreview()
        {
            _previewOn = !_previewOn;
            _preview.gameObject.SetActive(_previewOn);
            _previewLabel.text = _previewOn ? "[테스트] 전체 트리 미리보기: 켬" : "[테스트] 전체 트리 미리보기: 끔";
            StopAllCoroutines();
            StartCoroutine(Glide(_previewOn ? Vector2.zero : homeOffset, _previewOn ? 0.38f : 1f));
        }
