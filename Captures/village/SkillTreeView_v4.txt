    /// <summary>
    /// 노드형 스킬트리 화면.
    ///
    ///   위 — 제목, 보유 자원 6종
    ///   가운데 — 금속 링 원형 노드. 선행 노드를 찍어야 다음 노드가 열린다 (잠긴 노드는 자물쇠)
    ///   노드에 올리면 오른쪽에 툴팁 (갈래 · 이름 · 레벨 · 효과 · 비용 보유/필요)
    ///   아래 — [중앙 정렬] [테스트 미리보기] [마을로]
    ///
    /// 마을의 성장 관리 NPC가 연다. Esc나 [마을로]로 닫으면 마을로 돌아간다.
    ///
    /// 드래그로 이동, 휠로 확대/축소, 클릭하면 구매.
    /// 화면은 전부 코드로 만든다. 노드는 Progression.Nodes 데이터에서 읽는다.
    /// </summary>
    public class SkillTreeView : MonoBehaviour
    {
        [SerializeField] RunManager run;
        [SerializeField] IrisTransition iris;

        [Header("보기")]
        [SerializeField] float minZoom = 0.55f;
        [SerializeField] float maxZoom = 1.8f;
        [SerializeField] Vector2 homeOffset = new Vector2(0f, -250f);   // 트리가 위로 뻗으므로 시작점을 아래에 둔다

        /// <summary>노드 데이터 좌표에 곱하는 간격. 노드가 커진 만큼 벌린다.</summary>
        const float Spread = 1.35f;
        const float NodeSize = 108f, KeySize = 124f, CoreSize = 144f;
        const float PreviewSize = 84f, PreviewKeySize = 100f;

        // ── 색 ──
        static readonly Color ColBg = new Color(0.10f, 0.07f, 0.15f, 1f);
        static readonly Color ColGold = new Color(1f, 0.80f, 0.42f, 1f);
        static readonly Color ColGoldDim = new Color(0.72f, 0.56f, 0.32f, 1f);
        static readonly Color ColGoldHot = new Color(1f, 0.86f, 0.50f, 1f);
        static readonly Color ColSteel = new Color(0.46f, 0.44f, 0.54f, 1f);
        static readonly Color ColInner = new Color(0.08f, 0.06f, 0.12f, 1f);
        static readonly Color ColEdgeOn = new Color(0.98f, 0.74f, 0.36f, 1f);
        static readonly Color ColEdgeOff = new Color(0.36f, 0.34f, 0.44f, 1f);
        static readonly Color ColPanel = new Color(0.08f, 0.06f, 0.12f, 0.97f);
        static readonly Color ColBorder = new Color(0.86f, 0.64f, 0.30f, 1f);
        static readonly Color ColMuted = new Color(0.62f, 0.59f, 0.70f, 1f);

        class NodeView
        {
            public UpgradeNode Data;
            public RectTransform Root;
            public Image Glow, Ring, Inner, Art;
            public GameObject Lock;
            public TextMeshProUGUI Icon, Label;
            public readonly List<Image> InEdges = new List<Image>();
            public bool Preview;
            public bool Core;
        }

        RectTransform _root, _content, _edges, _nodes, _tooltip, _preview;
        TextMeshProUGUI _tipCat, _tipName, _tipBody, _tipCost, _tipStatus, _previewLabel;
        Image _tipCostIcon, _tipArrow;
        RectTransform _tipCostRow, _tipDiv2;
        Canvas _canvas;
        float _zoom = 1f;
        bool _previewOn;
        NodeView _hover;
        readonly Dictionary<NodeId, NodeView> _views = new Dictionary<NodeId, NodeView>();
        readonly Dictionary<ResourceId, TextMeshProUGUI> _bank = new Dictionary<ResourceId, TextMeshProUGUI>();

        public bool IsOpen => _root != null && _root.gameObject.activeSelf;
        float MinZoom => _previewOn ? 0.25f : minZoom;

        void Awake()
        {
            _canvas = GetComponentInParent<Canvas>();
            Build();
            _root.gameObject.SetActive(false);
        }

        void Start()
        {
            run.Bank.Changed += (_, __) => { if (IsOpen) Refresh(); };
            run.Progression.Changed += () => { if (IsOpen) Refresh(); };
        }

        public void Open()
        {
            _root.gameObject.SetActive(true);
            transform.SetAsLastSibling();
            if (_previewOn) SetPreview(false, false);
            _zoom = 1f;
            _content.localScale = Vector3.one;
            _content.anchoredPosition = homeOffset;
            HideTip();
            Refresh();
        }

        /// <summary>창을 닫은 순간. 마을이 플레이어 조작을 돌려준다.</summary>
        public event Action Closed;

        public void Close()
        {
            bool was = IsOpen;
            HideTip();
            _root.gameObject.SetActive(false);
            if (was) Closed?.Invoke();
        }

        // ───────────────────────────── 조립 ─────────────────────────────

        void Build()
        {
            _root = UiKit.Stretch(UiKit.Rect(transform, "SkillTree"));

            var bg = UiKit.Image(_root, "Background", ColBg, null, true);
            UiKit.Stretch(bg.rectTransform);
            bg.gameObject.AddComponent<TreeDragArea>().View = this;

            // 보라빛 성운 — 고정 시드라 매번 같다
            var rng = new System.Random(3);
            for (int i = 0; i < 7; i++)
            {
                var blob = UiKit.Image(bg.transform, "Nebula", new Color(0.38f, 0.20f, 0.58f, 0.10f + (float)rng.NextDouble() * 0.08f), UiKit.Soft);
                UiKit.Place(blob.rectTransform, new Vector2((float)rng.NextDouble(), (float)rng.NextDouble()),
                    Vector2.zero, Vector2.one * (500f + (float)rng.NextDouble() * 700f));
            }

            _content = UiKit.Rect(_root, "Graph");
            UiKit.Place(_content, new Vector2(0.5f, 0.5f), homeOffset, Vector2.zero);
            _edges = UiKit.Rect(_content, "Edges");
            _nodes = UiKit.Rect(_content, "Nodes");

            var core = MakeNode(_nodes, "Core", Vector2.zero, CoreSize);
            core.Core = true;
            core.Icon.text = "시작";
            SetArt(core, "explore_start");
            core.Ring.color = ColGoldHot;
            core.Glow.color = new Color(1f, 0.65f, 0.25f, 0.35f);

            foreach (var n in Progression.Nodes)
            {
                var v = MakeNode(_nodes, "Node_" + n.Id, n.Pos * Spread, n.Keystone ? KeySize : NodeSize);
                v.Data = n;
                v.Icon.text = n.Icon;
                SetArt(v, n.IconKey);
                Hook(v, () => TryBuy(v));
                _views[n.Id] = v;
            }

            foreach (var v in _views.Values)
            {
                if (v.Data.Requires.Length == 0) v.InEdges.Add(Edge(_edges, Vector2.zero, v.Data.Pos * Spread, 7f));
                foreach (var r in v.Data.Requires)
                    v.InEdges.Add(Edge(_edges, Progression.Find(r).Pos * Spread, v.Data.Pos * Spread, 7f));
            }

            BuildPreview();
            BuildHeader();
            BuildTooltip();
            BuildButtons();
        }

        NodeView MakeNode(Transform parent, string name, Vector2 pos, float size)
        {
            var v = new NodeView();
            v.Root = UiKit.Place(UiKit.Rect(parent, name), new Vector2(0.5f, 0.5f), pos, Vector2.one * size);

            v.Glow = UiKit.Image(v.Root, "Glow", new Color(1f, 0.6f, 0.2f, 0f), UiKit.Soft);
            UiKit.Place(v.Glow.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, Vector2.one * size * 1.9f);

            // 바깥 금속 링
            v.Ring = UiKit.Image(v.Root, "Ring", ColSteel, UiArt.MetalDisc);
            UiKit.Stretch(v.Ring.rectTransform);

            // 안쪽 어두운 판 + 아이콘 (원 안으로만 보이게 마스크)
            v.Inner = UiKit.Image(v.Root, "Inner", ColInner, UiArt.Disc, true);
            UiKit.Stretch(v.Inner.rectTransform);
            float inset = size * 0.075f;
            v.Inner.rectTransform.offsetMin = new Vector2(inset, inset);
            v.Inner.rectTransform.offsetMax = new Vector2(-inset, -inset);
            v.Inner.gameObject.AddComponent<Mask>().showMaskGraphic = true;

            v.Art = UiKit.Image(v.Inner.transform, "Art", Color.white);
            UiKit.Place(v.Art.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, Vector2.one * (size - inset * 2f) * 0.94f);
            v.Art.preserveAspect = true;
            v.Art.gameObject.SetActive(false);

            v.Icon = UiKit.Text(v.Root, "Icon", "", size * 0.3f);
            UiKit.Stretch(v.Icon.rectTransform);
            v.Icon.fontStyle = FontStyles.Bold;

            // 자물쇠 — 링 아래쪽에 걸친 작은 원
            var lockBg = UiKit.Image(v.Root, "Lock", new Color(0.12f, 0.11f, 0.17f, 1f), UiArt.Disc);
            UiKit.Place(lockBg.rectTransform, new Vector2(0.5f, 0f), new Vector2(0f, size * 0.06f), Vector2.one * size * 0.3f);
            var lockRim = UiKit.Image(lockBg.transform, "Rim", ColSteel, UiArt.MetalDisc);
            UiKit.Stretch(lockRim.rectTransform);
            lockRim.transform.SetAsFirstSibling();
            var lockIn = UiKit.Image(lockBg.transform, "In", new Color(0.12f, 0.11f, 0.17f, 1f), UiArt.Disc);
            UiKit.Stretch(lockIn.rectTransform);
            lockIn.rectTransform.offsetMin = new Vector2(2.5f, 2.5f);
            lockIn.rectTransform.offsetMax = new Vector2(-2.5f, -2.5f);
            var lockIcon = UiKit.Image(lockBg.transform, "Icon", new Color(0.9f, 0.88f, 0.95f, 1f), UiArt.Lock);
            UiKit.Place(lockIcon.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, Vector2.one * size * 0.19f);
            v.Lock = lockBg.gameObject;
            v.Lock.SetActive(false);

            v.Label = UiKit.Text(v.Root, "Label", "", 22);
            UiKit.Place(v.Label.rectTransform, new Vector2(0.5f, 0f), new Vector2(0f, -22f), new Vector2(220f, 28f));
            v.Label.color = new Color(0.9f, 0.88f, 0.94f, 1f);
            return v;
        }

        void Hook(NodeView v, Action click)
        {
            var hover = v.Inner.gameObject.AddComponent<TreeNodeHover>();
            hover.Enter = () => ShowTip(v);
            hover.Exit = () => { if (_hover == v) HideTip(); };
            hover.Click = click;
        }

        /// <summary>아이콘 그림을 붙인다. 파일이 없으면 한 글자를 그대로 둔다.</summary>
        static void SetArt(NodeView v, string key)
        {
            if (string.IsNullOrEmpty(key)) return;
            var sprite = Resources.Load<Sprite>("Icons/" + key);
            if (sprite == null) return;
            v.Art.sprite = sprite;
            v.Art.gameObject.SetActive(true);
            v.Icon.gameObject.SetActive(false);
        }

        static Image Edge(Transform parent, Vector2 a, Vector2 b, float width)
        {
            var img = UiKit.Image(parent, "Edge", ColEdgeOff);
            var rt = img.rectTransform;
            Vector2 d = b - a;
            UiKit.Place(rt, new Vector2(0.5f, 0.5f), (a + b) * 0.5f, new Vector2(d.magnitude, width));
            rt.localRotation = Quaternion.Euler(0f, 0f, Mathf.Atan2(d.y, d.x) * Mathf.Rad2Deg);
            return img;
        }

        // ───────────────────────────── 상태 ─────────────────────────────

        void Refresh()
        {
            var prog = run.Progression;
            foreach (var v in _views.Values)
            {
                var n = v.Data;
                int lv = prog.Level(n.Id);
                bool unlocked = prog.IsUnlocked(n);
                bool maxed = prog.IsMaxed(n);
                bool afford = prog.CanBuy(n, run.Bank);

                v.Lock.SetActive(!unlocked);
                v.Ring.color = !unlocked ? ColSteel : afford ? ColGoldHot : lv > 0 ? ColGold : ColGoldDim;
                v.Art.color = !unlocked ? new Color(0.36f, 0.34f, 0.42f, 1f)
                            : (lv > 0 || afford) ? Color.white : new Color(0.78f, 0.78f, 0.80f, 1f);
                v.Icon.color = unlocked ? Color.white : new Color(1f, 1f, 1f, 0.3f);

                if (!unlocked) v.Label.text = "";
                else if (n.MaxLevel > 1) v.Label.text = maxed ? "<color=#ffd873>MAX</color>" : $"{lv}/{n.MaxLevel}";
                else v.Label.text = lv > 0 ? "<color=#ffd873>습득</color>" : "";

                bool lit = n.Requires.Length == 0 || unlocked;
                foreach (var e in v.InEdges) e.color = lit ? ColEdgeOn : ColEdgeOff;
            }

            foreach (var kv in _bank) kv.Value.text = run.Bank.Get(kv.Key).ToString();
            if (_hover != null) ShowTip(_hover);
        }

        void Update()
        {
            if (!IsOpen) return;
            var kb = Keyboard.current;
            if (kb != null && kb.escapeKey.wasPressedThisFrame && (iris == null || !iris.Busy)) { Close(); return; }
            // 살 수 있는 노드는 은은하게 숨 쉰다 — 어디를 눌러야 하는지 한눈에 보인다
            float pulse = 0.28f + 0.2f * Mathf.Sin(Time.unscaledTime * 4f);
            foreach (var v in _views.Values)
            {
                var g = v.Glow.color;
                g.a = run.Progression.CanBuy(v.Data, run.Bank) ? pulse : 0f;
                v.Glow.color = g;
            }
        }

        void TryBuy(NodeView v)
        {
            if (!run.Progression.Buy(v.Data, run.Bank)) { StartCoroutine(Shake(v.Root)); return; }
            StartCoroutine(Pop(v.Root));
        }

        IEnumerator Pop(RectTransform rt)
        {
            float t = 0f;
            while (t < 0.25f)
            {
                t += Time.unscaledDeltaTime;
                float k = t / 0.25f;
                rt.localScale = Vector3.one * (1f + 0.22f * Mathf.Sin(k * Mathf.PI) * (1f - k * 0.5f));
                yield return null;
            }
            rt.localScale = Vector3.one;
        }

        IEnumerator Shake(RectTransform rt)
        {
            Vector2 p0 = rt.anchoredPosition;
            float t = 0f;
            while (t < 0.22f)
            {
                t += Time.unscaledDeltaTime;

        static (string, Color) Category(string key)
        {
            if (string.IsNullOrEmpty(key)) return ("", Color.white);
            if (key.StartsWith("explore")) return ("탐험", new Color(0.55f, 0.92f, 0.45f));
            if (key.StartsWith("mine")) return ("채굴", new Color(0.45f, 0.85f, 1f));
            if (key.StartsWith("combat")) return ("전투", new Color(1f, 0.55f, 0.38f));
            if (key.StartsWith("hire")) return ("고용", new Color(1f, 0.85f, 0.35f));
            if (key.StartsWith("skill")) return ("스킬", new Color(0.82f, 0.72f, 1f));
            return ("", Color.white);
        }

        void ShowTip(NodeView v)
        {
            if (v.Core) return;
            _hover = v;
            var n = v.Data;
            var (cat, catCol) = Category(n.IconKey);
            _tipCat.text = cat;
            _tipCat.color = catCol;

            if (v.Preview)
            {
                _tipName.text = n.Name;
                _tipBody.text = n.Effect;
                _tipCostRow.gameObject.SetActive(false);
                _tipStatus.text = "미리보기 · 아직 구현되지 않은 노드";
                _tipStatus.color = ColMuted;
                PlaceTip(v);
                return;
            }

            var prog = run.Progression;
            int lv = prog.Level(n.Id);
            bool maxed = prog.IsMaxed(n);
            bool unlocked = prog.IsUnlocked(n);

            _tipName.text = n.MaxLevel > 1
                ? $"{n.Name}   <size=22><color=#9a94ab>Lv. {lv} / {n.MaxLevel}</color></size>"
                : n.Name;

            string body = n.Effect;
            if (n.Total != null && n.MaxLevel > 1)
                body += "\n<size=18><color=#9a94ab>" + (maxed ? "현재 " + n.Total(lv) : $"{n.Total(lv)}  →  {n.Total(lv + 1)}") + "</color></size>";
            _tipBody.text = body;

            if (maxed)
            {
                _tipCostRow.gameObject.SetActive(false);
                _tipStatus.text = "최대 레벨";
                _tipStatus.color = ColGold;
            }
            else
            {
                _tipCostRow.gameObject.SetActive(true);
                int have = run.Bank.Get(n.Cost), need = prog.NextCost(n);
                bool enough = have >= need;
                _tipCostIcon.sprite = Resources.Load<Sprite>("Icons/res_" + n.Cost.ToString().ToLowerInvariant());
                _tipCost.text = $"<color={(enough ? "#9fe8a6" : "#ff6b6b")}>{have} / {need}</color>";

                if (!unlocked)
                {
                    var req = new System.Text.StringBuilder();
                    foreach (var r in n.Requires) req.Append(Progression.Find(r).Name).Append(' ');
                    _tipStatus.text = "선행 노드 필요: " + req.ToString().Trim();
                    _tipStatus.color = new Color(1f, 0.55f, 0.5f);
                }
                else if (!enough)
                {
                    _tipStatus.text = "자원이 부족합니다";
                    _tipStatus.color = ColMuted;
                }
                else
                {
                    _tipStatus.text = "클릭해서 배우기";
                    _tipStatus.color = new Color(0.62f, 0.92f, 0.66f);
                }
            }
            PlaceTip(v);
        }

        /// <summary>노드 오른쪽에 띄운다. 화면 밖으로 나가면 왼쪽으로 뒤집는다.</summary>
        void PlaceTip(NodeView v)
        {
            _tooltip.gameObject.SetActive(true);
            _tooltip.SetAsLastSibling();

            float half = v.Root.rect.width * 0.5f * v.Root.lossyScale.x / Mathf.Max(0.0001f, _root.lossyScale.x);
            Vector2 center;
            RectTransformUtility.ScreenPointToLocalPointInRectangle(_root,
                RectTransformUtility.WorldToScreenPoint(null, v.Root.position), null, out center);

            float gap = 22f;
            bool flip = center.x + half + gap + _tooltip.sizeDelta.x > _root.rect.width * 0.5f - 20f;
            _tooltip.pivot = new Vector2(flip ? 1f : 0f, 0.5f);
            _tooltip.anchoredPosition = center + new Vector2(flip ? -(half + gap) : (half + gap), 0f);

            // 위아래로 넘치지 않게
            var p = _tooltip.anchoredPosition;
            float hh = _tooltip.sizeDelta.y * 0.5f, lim = _root.rect.height * 0.5f - 20f;
            p.y = Mathf.Clamp(p.y, -lim + hh, lim - hh);
            _tooltip.anchoredPosition = p;

            // 꼬리는 노드 쪽 가장자리, 노드 높이에 맞춘다
            var art = _tipArrow.rectTransform;
            art.anchorMin = art.anchorMax = new Vector2(flip ? 1f : 0f, 0.5f);
            art.sizeDelta = new Vector2(26f, 26f);
            art.anchoredPosition = new Vector2(flip ? 11f : -11f, Mathf.Clamp(center.y - p.y, -hh + 24f, hh - 24f));
            art.localRotation = Quaternion.Euler(0f, 0f, flip ? 180f : 0f);
        }

        void HideTip()
        {
            _hover = null;
            if (_tooltip != null) _tooltip.gameObject.SetActive(false);
        }

        void Recenter()
        {
            StopAllCoroutines();
            StartCoroutine(Glide(_previewOn ? new Vector2(0f, 60f) : homeOffset, _previewOn ? 0.3f : 1f));
        }

        IEnumerator Glide(Vector2 to, float zoom)
        {
            HideTip();
            Vector2 from = _content.anchoredPosition;
            float z0 = _zoom, t = 0f;
            while (t < 0.3f)
            {
                t += Time.unscaledDeltaTime;
                float k = 1f - Mathf.Pow(1f - Mathf.Clamp01(t / 0.3f), 3f);
                _content.anchoredPosition = Vector2.Lerp(from, to, k);
                _zoom = Mathf.Lerp(z0, zoom, k);
                _content.localScale = Vector3.one * _zoom;
                yield return null;
            }
        }

        internal void OnDragged(Vector2 screenDelta)
        {
            HideTip();
            float s = _canvas != null ? _canvas.scaleFactor : 1f;
            _content.anchoredPosition += screenDelta / s;
        }

        internal void OnScrolled(Vector2 screenPos, float delta)
        {
            if (Mathf.Abs(delta) < 0.01f) return;
            HideTip();
            float newZoom = Mathf.Clamp(_zoom * (delta > 0 ? 1.12f : 1f / 1.12f), MinZoom, maxZoom);
            RectTransformUtility.ScreenPointToLocalPointInRectangle(_content, screenPos, null, out var before);
            _zoom = newZoom;
            _content.localScale = Vector3.one * _zoom;
            RectTransformUtility.ScreenPointToLocalPointInRectangle(_content, screenPos, null, out var after);
            _content.anchoredPosition += (after - before) * _zoom;
        }

        /// <summary>
        /// 갈래별로 줄지어 배치한다. 위·아래 갈래는 좌우 2칸, 좌·우 갈래는 위아래 4칸.
        /// 가운데 축은 실제 노드 자리로 비워 둔다.
        /// </summary>
        void BuildPreview()
        {
            _preview = UiKit.Rect(_content, "Preview");
            _preview.SetSiblingIndex(0);   // 실제 노드 아래에 깔린다
            var edges = UiKit.Rect(_preview, "Edges");
            var nodes = UiKit.Rect(_preview, "Nodes");

            var count = new Dictionary<string, int>();
            foreach (var d in PreviewData)
            {
                string br = d[0];
                count.TryGetValue(br, out int i);
                count[br] = i + 1;

                Vector2 dir, side; float start; float[] slots;
                switch (br)
                {
                    case "E": dir = Vector2.up; side = Vector2.right; start = 150f; slots = new[] { -150f, 150f }; break;
                    case "H": dir = Vector2.down; side = Vector2.right; start = 150f; slots = new[] { -150f, 150f }; break;
                    case "M": dir = Vector2.right; side = Vector2.up; start = 340f; slots = new[] { -150f, 150f, -300f, 300f }; break;
                    default: dir = Vector2.left; side = Vector2.up; start = 340f; slots = new[] { -150f, 150f, -300f, 300f }; break;
                }
                int row = i / slots.Length;
                float along = start + row * 150f;
                Vector2 pos = (dir * along + side * slots[i % slots.Length]) * Spread;
                Vector2 anchor = dir * (along - 150f) * Spread;   // 앞 줄의 축 지점에서 뻗는다

                var e = Edge(edges, anchor, pos, 4f);
                e.color = ColEdgeOff * new Color(1f, 1f, 1f, 0.7f);

                bool key = d[4] == "1";
                var v = MakeNode(nodes, "Preview_" + d[1], pos, key ? PreviewKeySize : PreviewSize);
                v.Preview = true;
                v.Data = new UpgradeNode { Name = d[1], Effect = d[2], MaxLevel = 1, Keystone = key, IconKey = d[3] };
                v.Icon.text = d[1].Substring(0, 1);
                SetArt(v, d[3]);
                v.Ring.color = key ? ColGoldDim : ColSteel;
                v.Art.color = new Color(0.66f, 0.66f, 0.72f, 1f);
                v.Label.text = "<color=#8f8aa3>" + d[1] + "</color>";
                v.Label.fontSize = 17;
                var captured = v;
                Hook(v, () => StartCoroutine(Shake(captured.Root)));
            }

            // 갈래 축 — 줄 사이를 잇는 흐린 선
            foreach (var kv in count)
            {
                Vector2 dir = kv.Key == "E" ? Vector2.up : kv.Key == "H" ? Vector2.down : kv.Key == "M" ? Vector2.right : Vector2.left;
                int per = (kv.Key == "E" || kv.Key == "H") ? 2 : 4;
                float start = (kv.Key == "E" || kv.Key == "H") ? 150f : 340f;
                float end = (start + ((kv.Value - 1) / per - 1) * 150f) * Spread;
                if (end <= 0f) continue;
                var e = Edge(edges, Vector2.zero, dir * end, 4f);
                e.color = ColEdgeOff * new Color(1f, 1f, 1f, 0.7f);
                e.transform.SetAsFirstSibling();
            }

            _preview.gameObject.SetActive(false);
        }

        void SetPreview(bool on, bool glide)
        {
            _previewOn = on;
            _preview.gameObject.SetActive(on);
            if (_previewLabel != null)
                _previewLabel.text = on ? "[테스트] 미리보기: 켬" : "[테스트] 미리보기";
            HideTip();
            if (!glide) return;
            StopAllCoroutines();
            StartCoroutine(Glide(on ? new Vector2(0f, 60f) : homeOffset, on ? 0.3f : 1f));
        }
    }

    }

    /// <summary>노드 위에 올렸을 때 / 뗐을 때 / 눌렀을 때.</summary>
    public class TreeNodeHover : MonoBehaviour, IPointerEnterHandler, IPointerExitHandler, IPointerClickHandler, IScrollHandler, IDragHandler
    {
        public Action Enter, Exit, Click;
        public void OnPointerEnter(PointerEventData e) => Enter?.Invoke();
        public void OnPointerExit(PointerEventData e) => Exit?.Invoke();
        public void OnPointerClick(PointerEventData e) { if (!e.dragging) Click?.Invoke(); }

        // 노드 위에서도 휠과 드래그가 먹도록 배경으로 넘긴다
        public void OnScroll(PointerEventData e) => ExecuteEvents.ExecuteHierarchy(transform.root.GetComponentInChildren<TreeDragArea>().gameObject, e, ExecuteEvents.scrollHandler);
        public void OnDrag(PointerEventData e) => ExecuteEvents.ExecuteHierarchy(transform.root.GetComponentInChildren<TreeDragArea>().gameObject, e, ExecuteEvents.dragHandler);
    }
