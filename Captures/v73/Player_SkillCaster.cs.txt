        // 커맨드 기록 — 절대 방향(L R U D)과 그때 바라보던 쪽
        struct Token { public char Dir; public float At; public bool FacingRight; }
        readonly List<Token> _tokens = new();
        Vector2Int _lastDir;

        void Awake()
        {
            _pc = GetComponent<PlayerController>();
            _cc = GetComponent<CharacterController>();
            _health = GetComponent<Health>();
            _anim = GetComponentInChildren<Animator>();
            foreach (var s in Specs) s.Book = Array.FindIndex(SkillBook.All, d => d.Name == s.Name);
            _health.Intercept += OnIncoming;
        }

        void OnDestroy() { if (_health != null) _health.Intercept -= OnIncoming; }

        RunManager Run => _run != null ? _run : (_run = FindFirstObjectByType<RunManager>());

        void Update()
        {
            for (int i = 0; i < _cdLeft.Length; i++) if (_cdLeft[i] > 0f) _cdLeft[i] -= Time.deltaTime;
            RecordDirections();
            if (_cur != null) Tick();
        }

        void RecordDirections()
        {
            var v = GameInput.MoveValue;
            Vector2Int d = Vector2Int.zero;
            if (v.sqrMagnitude > 0.35f)
                d = Mathf.Abs(v.x) >= Mathf.Abs(v.y) ? new Vector2Int(v.x > 0 ? 1 : -1, 0) : new Vector2Int(0, v.y > 0 ? 1 : -1);
            if (d != _lastDir && d != Vector2Int.zero)
            {
                _tokens.Add(new Token { Dir = d.x > 0 ? 'R' : d.x < 0 ? 'L' : d.y > 0 ? 'U' : 'D', At = Time.unscaledTime, FacingRight = _pc.FacingRight });
                if (_tokens.Count > 6) _tokens.RemoveAt(0);
            }
            _lastDir = d;
        }

        // ───────────────────────────── 발동 ─────────────────────────────

        /// <summary>
        /// PlayerController가 매 프레임 먼저 묻는다. 스킬을 냈으면 true — 그 프레임의 평타·이동은 하지 않는다.
        /// 평타 도중에도 낼 수 있다 (스킬 캔슬).
        /// </summary>
        public bool TryCast(Vector2 move)
        {
            if (_cur != null) return true;
            var run = Run;
            if (run == null || run.Phase != RunPhase.InDungeon) return false;

            // 퀵슬롯 — 커맨드 없이
            int slot = GameInput.SlotPressed();
            if (slot >= 0)
            {
                int k = run.Skills.Slot(slot);
                if (k >= 0)
                {
                    if (Mathf.Abs(move.x) > 0.3f) _pc.SetFacing(move.x > 0f);
                    return Begin(SpecOf(k));
                }
            }

            bool z = GameInput.SkillPressed, x = GameInput.AttackPressed;
            if (!z && !x) return false;
            char button = z ? 'Z' : 'X';

            var spec = Match(button, out bool face);
            if (spec != null && Begin(spec)) { _pc.SetFacing(face); _tokens.Clear(); return true; }
            if (z) return Begin(SpecOf(Array.FindIndex(SkillBook.All, d => d.Cmd == "Z")));   // Z 단독 = 올려베기
            return false;   // X = 평타
        }

        /// <summary>커맨드에 맞는 스킬. 긴 커맨드가 이긴다. face = 스킬을 쓸 방향.</summary>
        Spec Match(char button, out bool face)
        {
            face = _pc.FacingRight;
            float window = Mathf.Max(0.2f, GameSettings.Data.commandWindow);
            float now = Time.unscaledTime;

            // 끝에서부터 간격이 여유 안인 토큰만 이어 붙인다
            int start = _tokens.Count;
            float t = now;
            for (int i = _tokens.Count - 1; i >= 0; i--)
            {
                if (t - _tokens[i].At > window) break;
                t = _tokens[i].At; start = i;
            }
            if (start >= _tokens.Count) return null;

            bool firstFacing = _tokens[start].FacingRight;
            var s = Best(button, Relative(start, firstFacing));
            if (s != null) { face = firstFacing; return s; }
            s = Best(button, Relative(start, _pc.FacingRight));
            if (s != null) { face = _pc.FacingRight; return s; }
            return null;
        }

        string Relative(int start, bool facingRight)
        {
            var sb = new System.Text.StringBuilder();
            for (int i = start; i < _tokens.Count; i++)
            {
                char c = _tokens[i].Dir;
                sb.Append(c == 'U' ? '↑' : c == 'D' ? '↓' : (c == 'R') == facingRight ? '→' : '←');
            }
            return sb.ToString();
        }

        Spec Best(char button, string seq)
        {
            Spec best = null; int bestLen = 0;
            var book = Run.Skills;
            foreach (var s in Specs)
            {
                if (s.Book < 0 || !book.Learned(s.Book)) continue;
                var cmd = SkillBook.All[s.Book].Cmd.Replace(" ", "");
                if (cmd.Length < 2 || cmd[cmd.Length - 1] != button) continue;
                var dirs = cmd.Substring(0, cmd.Length - 1);
                if (dirs.Length > bestLen && seq.EndsWith(dirs, StringComparison.Ordinal)) { best = s; bestLen = dirs.Length; }
            }
            return best;
        }

        /// <summary>지금 레벨 기준 재사용 시간 (스킬창 설명이 쓴다).</summary>
        public float CooldownOf(int book)
        {
            var s = SpecOf(book);
            if (s == null || Run == null) return 0f;
            int lv = Mathf.Max(1, Run.Skills.Level(book));
            return Mathf.Max(1f, s.Cooldown + s.CooldownPerLevel * (lv - 1));
        }

        /// <summary>재사용 시간 배율 — 스킬 테스트 씬에서 0 (재사용 없음).</summary>
        public static float CooldownScale = 1f;

        Spec SpecOf(int book)
        {
            foreach (var s in Specs) if (s.Book == book) return s;
            return null;
        }

        bool Begin(Spec s)
        {
            if (s == null || s.Book < 0 || !Run.Skills.Learned(s.Book)) return false;
            if (_cdLeft[s.Book] > 0f) return false;   // 재사용 중 — X 커맨드면 평타가 대신 나간다

            _curLevel = Run.Skills.Level(s.Book);
            float cd = Mathf.Max(1f, s.Cooldown + s.CooldownPerLevel * (_curLevel - 1)) * CooldownScale;
            _cdLeft[s.Book] = cd;
            int slot = SlotOf(s.Book);
            if (slot >= 0 && Proto.UI.SkillBarView.Instance != null) Proto.UI.SkillBarView.Instance.Use(slot, cd);

            var feel = Proto.Feel.Feel.I;
            var head = transform.position + Vector3.up * 2.3f;

            // 모션 없는 버프
            if (s.Kind == Kind.Sprint)
            {
                _buffSpeedUntil = Time.time + SprintBase + SprintPerLevel * (_curLevel - 1);
                if (feel != null) feel.Popup(head, "질주");
                return true;
            }

            _cur = s;
            _t = s.Offset;
            _nextHit = 0;
            _countered = false;
            _guardPhase = 0;
            _pc.BeginSkill();
            Play(s.State, s.Offset);
            if (feel != null) feel.SwingSfx();
            // 회전베기 — 몸 둘레를 도는 궤적 · 발밑 소용돌이 · 흙먼지 (휠윈드)
            // 스킬 이펙트 — 배정표(Feel/SkillVfx)에 있는 스킬은 산 이펙트 팩으로, 없으면 예전 직접 만든 것
            if (Proto.Feel.SkillVfx.Has(s.Name)) Proto.Feel.SkillVfx.Play(s.Name, Proto.Feel.SkillVfx.Start, transform, _pc.FacingRight);
            else if (s.Name == "회전베기") Proto.Feel.SkillFx.Whirl(transform, (s.End - s.Offset) / Mathf.Max(0.1f, s.Speed), 2.6f, !_pc.FacingRight);
            return true;
        }

        int SlotOf(int book)
        {
            for (int i = 0; i < SkillBook.SlotCount; i++) if (Run.Skills.Slot(i) == book) return i;
            return -1;
        }

        void Play(string state, float offset)
        {
            if (_anim == null || string.IsNullOrEmpty(state)) return;
            _anim.CrossFadeInFixedTime(state, 0.06f, 0, offset);
        }

        /// <summary>죽거나 조작이 막히면 PlayerController가 부른다.</summary>
        public void Cancel()
        {
            if (_cur == null) return;
            _cur = null;
            _pc.EndSkill();
        }

        void Finish()
        {
            _cur = null;
            if (_anim != null) _anim.CrossFadeInFixedTime("Idle", 0.2f);
            _pc.EndSkill();
        }

        // ───────────────────────────── 진행 ─────────────────────────────

        void Tick()
        {
            var s = _cur;
            if (s.Kind == Kind.Counter) { TickCounter(); return; }

            float prev = _t;
            _t += Time.deltaTime * s.Speed;

            // 이동 — 정해진 구간에 앞으로
            float dist = s.MoveDist + s.MoveDistPerLevel * (_curLevel - 1);
            float move = 0f;
            if (dist > 0f && s.MoveTo > s.MoveFrom)
            {
                float a = Mathf.Clamp01(Mathf.InverseLerp(s.MoveFrom, s.MoveTo, prev));
                float b = Mathf.Clamp01(Mathf.InverseLerp(s.MoveFrom, s.MoveTo, _t));
                move = dist * (b - a);
            }
            _cc.Move(Forward * move + Vector3.down * 2f * Time.deltaTime);

            while (_nextHit < s.Hits.Length && _t >= s.Hits[_nextHit].At)
            {
                var h = s.Hits[_nextHit++];
                if (_curLevel < h.MinLevel) continue;
                Fire(s, h);
            }

            if (s.Kind == Kind.WarCry && prev < 1.3f && _t >= 1.3f) WarCry();

            if (_t >= s.End) Finish();
        }

        Vector3 Forward => _pc.FacingRight ? Vector3.right : Vector3.left;

        void Fire(Spec s, Hit h)
        {
            switch (s.Kind)
            {
                case Kind.Wave: SpawnWave(s, h); break;
                default: Strike(s, h); break;
            }
        }

        // ── 판정 ──

        /// <summary>판정 하나. 맞힌 수를 돌려준다.</summary>
        int Strike(Spec s, Hit h, Vector3? origin = null, HashSet<Health> once = null)
        {
            var o = origin ?? transform.position;
            var fwd = Forward;
            var cfg = _pc.Config;
            var feel = Proto.Feel.Feel.I;

            // 칼 지나간 자리
            if (origin == null && Proto.Feel.SkillVfx.Has(s.Name))
            {
                int hitIndex = s.Kind == Kind.Counter ? 0 : _nextHit - 1;
                Proto.Feel.SkillVfx.Play(s.Name, hitIndex, transform, _pc.FacingRight);
                if (h.Heavy) Proto.Feel.HitFx.Impact(o + fwd * (h.Shape == Shape.Around ? 0f : 1.4f) + Vector3.up * 0.2f, true);
            }
            else if (origin == null)
            {
                Vector3 swing = o + fwd * 1.0f + Vector3.up * 1.0f;
                if (h.Shape == Shape.Around)
                {
                    Proto.Feel.HitFx.Slash(o + Vector3.right * 1.1f + Vector3.up * 0.9f, true, _nextHit % 2, h.Reach * 1.2f);
                    Proto.Feel.HitFx.Slash(o + Vector3.left * 1.1f + Vector3.up * 0.9f, false, (_nextHit + 1) % 2, h.Reach * 1.2f);
                }
                else if (s.Name == "연속 찌르기")   // 펜싱처럼 — 칼끝에서 앞으로 빛창이 쭉 뻗는다
                    Proto.Feel.SkillFx.Thrust(o + fwd * 0.5f + Vector3.up * (1.05f + UnityEngine.Random.Range(-0.15f, 0.15f)), fwd, h.Reach + 0.4f);
                else Proto.Feel.HitFx.Slash(swing, _pc.FacingRight, h.Heavy ? 2 : _nextHit % 2, h.Heavy ? 4f : 3.2f);
                if (h.Heavy) Proto.Feel.HitFx.Impact(o + fwd * (h.Shape == Shape.Around ? 0f : 1.4f) + Vector3.up * 0.2f, true);
            }

            float range = h.Shape == Shape.Around ? h.Reach : h.Reach + 0.5f;
            var cols = Physics.OverlapCapsule(o, o + Vector3.up * 3.2f, range + 1f, _pc.EnemyMask);   // 위로 길게 — 떠 있는 적도 맞는다
            var struck = new HashSet<Health>();
            float dmgBase = cfg.attackDamage * h.Mul * (1f + s.DamagePerLevel * (_curLevel - 1)) * DamageMul;
            int connected = 0; bool killed = false;

            foreach (var c in cols)
            {
                var hp = c.GetComponentInParent<Health>();
                if (hp == null || hp.IsDead || hp.gameObject == gameObject || !struck.Add(hp)) continue;
                if (once != null && once.Contains(hp)) continue;

                var to = hp.transform.position - o; to.y = 0f;
                if (h.Shape == Shape.Around) { if (to.magnitude > h.Reach) continue; }
                else
                {
                    float along = Vector3.Dot(to, fwd);
                    if (along < -0.4f || along > h.Reach || Mathf.Abs(to.z) > h.Depth) continue;
                }
                if (once != null) once.Add(hp);   // 검기 — 범위 안에 들어온 뒤에야 '맞은 적'으로 친다

                float dmg = dmgBase;
                var ec = hp.GetComponent<Proto.Enemy.EnemyController>();
                if (ec != null) dmg = ec.ModifyIncomingDamage(dmg);
                var boss = hp.GetComponent<Proto.Enemy.BossController>();
                if (boss != null) dmg = boss.ModifyIncomingDamage(dmg);
                dmg = CombatMath.Crit(cfg, dmg, out bool crit);   // 장비 치명타

                hp.TakeDamage(dmg, o);
                connected++;
                bool kill = hp.IsDead;
                killed |= kill;

                // 밀기 · 띄우기 — 기본 넉백을 덮어쓴다 (슈퍼아머는 무시)
                if (ec != null)
                {
                    var dir = h.Shape == Shape.Around && to.sqrMagnitude > 0.01f ? to.normalized : fwd;
                    ec.ApplyHit(dir * (h.Knock / Mathf.Max(0.1f, h.Stun)), h.Stun, h.Lift);
                }

                if (feel != null)
                {
                    var at = hp.transform.position + Vector3.up * 0.9f - fwd * 0.3f;
                    feel.Spark(at, fwd, kill || h.Heavy);
                    Proto.Feel.HitFx.Impact(at, kill || h.Heavy);
                    Proto.Feel.SkillVfx.Impact(at, _pc.FacingRight, kill || h.Heavy);
                    feel.Number(at + Vector3.up * 0.6f, dmg, kill ? Proto.Feel.Feel.NumberKind.Kill : crit ? Proto.Feel.Feel.NumberKind.Crit : Proto.Feel.Feel.NumberKind.Normal);
                }
            }

            if (feel != null && (connected > 0 || h.Heavy))
            {
                if (h.Heavy || killed)
                {
                    feel.HitStop(cfg.killHitStopScale);
                    feel.Shake(fwd, h.Mul >= 5f ? 2.4f : 1.5f);
                    Proto.Feel.ScreenPunch.Heavy();   // 색수차 + 렌즈 펀치 (MMFeedbacks)
                    if (connected > 0) feel.KillSfx();
                    Proto.Feel.Haptics.Pulse(0.45f, 0.75f, h.Mul >= 5f ? 0.3f : 0.15f);
                }
                else
                {
                    feel.HitStop(0.7f);
                    feel.Shake(fwd, 0.6f);
                    Proto.Feel.ScreenPunch.Light();
                    feel.HitSfx();
                    Proto.Feel.Haptics.Pulse(0.15f, 0.3f, 0.05f);
                }
            }
            return connected;
        }

        // ── 검기 ──

        void SpawnWave(Spec s, Hit h)
        {
            if (!Proto.Feel.SkillVfx.Play(s.Name, 0, transform, _pc.FacingRight))
                Proto.Feel.HitFx.Slash(transform.position + Forward * 1.0f + Vector3.up * 1.0f, _pc.FacingRight, 2, 3.6f);
            var go = new GameObject("SwordWave");
            go.transform.position = transform.position + Vector3.up * 0.1f + Forward * 0.8f;
            var w = go.AddComponent<SwordWave>();
            float range = h.Reach + WaveRangePerLevel * (_curLevel - 1);
            w.Init(this, s, h, Forward, _pc.FacingRight, range / WaveSpeed);
        }

        /// <summary>검기가 날아가며 부른다 — 한 적은 한 번만.</summary>
        void WaveStrike(Spec s, Hit h, Vector3 at, HashSet<Health> once) => Strike(s, h, at, once);

        class SwordWave : MonoBehaviour
        {
            SkillCaster _owner; Spec _s; Hit _h; Vector3 _dir; bool _right; float _life, _fx; GameObject _trail; List<GameObject> _packFx;
            readonly HashSet<Health> _once = new();

            public void Init(SkillCaster owner, Spec s, Hit h, Vector3 dir, bool right, float life)
            { _owner = owner; _s = s; _h = new Hit { At = h.At, Mul = h.Mul, Reach = 1.2f, Depth = h.Depth, Knock = h.Knock, Stun = h.Stun }; _dir = dir; _right = right; _life = life; _packFx = Proto.Feel.SkillVfx.AttachWave(s.Name, transform, right);
              if (_packFx.Count == 0) _trail = Proto.Feel.SkillFx.WaveTrail(transform, right); }   // 이펙트 팩 검기 — 없으면 직접 만든 초승달

            void Update()
            {
                float dt = Time.deltaTime;
                transform.position += _dir * WaveSpeed * dt;
                _life -= dt;
                _fx -= dt;
                if (_owner != null) _owner.WaveStrike(_s, _h, transform.position - _dir * 0.6f, _once);
                if (_life <= 0f) { Proto.Feel.SkillFx.Release(_trail); Proto.Feel.SkillVfx.Release(_packFx); Destroy(gameObject); }
            }
        }

        // ── 반격 자세 ──

        void TickCounter()
        {
            _t += Time.deltaTime;
            switch (_guardPhase)
            {
                case 0:   // 자세 잡기
                    if (_t >= 0.3f)
                    {
                        _guardPhase = 1; _t = 0f;
                        _guardUntil = Time.time + CounterBase + CounterPerLevel * (_curLevel - 1);
                        Play("Skill_GuardLoop", 0f);
                    }
                    break;
                case 1:   // 버티기 — 이 동안 맞으면 OnIncoming이 막고 반격으로 넘긴다
                    if (_countered) { _guardPhase = 2; _t = 0f; _nextHit = 0; Play("Skill_GuardAtk", 0f); }
                    else if (Time.time >= _guardUntil) { _guardPhase = 3; _t = 0f; Play("Skill_GuardEnd", 0f); }
                    break;
                case 2:   // 반격
                    if (_nextHit == 0 && _t >= _cur.Hits[0].At) { _nextHit = 1; Strike(_cur, _cur.Hits[0]); }
                    if (_t >= 1.15f) Finish();
                    break;
                case 3:
                    if (_t >= 0.5f) Finish();
                    break;
            }
        }

        /// <summary>Health가 피해를 받기 직전에 묻는다. true면 막은 것.</summary>
        bool OnIncoming(float amount, Vector3 src)
        {
            if (_cur == null || _cur.Kind != Kind.Counter || _guardPhase != 1) return false;
            _countered = true;
            // 때린 쪽을 본다
            _pc.SetFacing(src.x > transform.position.x);
            var feel = Proto.Feel.Feel.I;
            if (feel != null)
            {
                feel.Popup(transform.position + Vector3.up * 2.3f, "반격!");
                feel.HitStop(1.6f);
                Proto.Feel.HitFx.Impact(transform.position + Forward * 0.6f + Vector3.up * 1.1f, true);
            }
            Proto.Feel.SkillVfx.Block(transform.position + Forward * 0.6f + Vector3.up * 1.1f);
            Proto.Feel.Haptics.Pulse(0.3f, 0.8f, 0.12f);
            return true;
        }

        // ── 버프 ──

        void WarCry()
        {
            _buffDamageUntil = Time.time + WarCryBase + WarCryPerLevel * (_curLevel - 1);
            var feel = Proto.Feel.Feel.I;
            if (feel != null)
            {
                feel.Popup(transform.position + Vector3.up * 2.5f, "공격력 +" + Mathf.RoundToInt(WarCryDamage * 100f) + "%");
                feel.Shake(Vector3.up, 1.2f);
            }
            Proto.Feel.HitFx.Impact(transform.position + Vector3.up * 1.6f, true);
            Proto.Feel.Haptics.Pulse(0.4f, 0.4f, 0.25f);
        }
    }
