        public static DungeonDef[] LoadAll()
        {
            var names = new[] { "01_Forest", "02_Mine", "03_SnowTemple", "04_Rift" };
            var l = new List<DungeonDef>();
            foreach (var n in names)
            {
                var d = AssetDatabase.LoadAssetAtPath<DungeonDef>(Dir + n + ".asset");
                if (d != null) l.Add(d);
            }
            return l.Count > 0 ? l.ToArray() : BuildAll();
        }

        /// <summary>씬의 마을에 던전 목록·도리이·선택 창을 잇는다. 선택 창이 없으면 만든다.</summary>
        public static void WireScene(DungeonDef[] list)
        {
            var village = Object.FindFirstObjectByType<VillageController>(FindObjectsInactive.Include);
            if (village == null) { Debug.LogWarning("[Proto] 씬에 VillageController가 없다"); return; }
            var gate = Object.FindFirstObjectByType<VillageGate>(FindObjectsInactive.Include);
            var canvas = Object.FindFirstObjectByType<SkillWindowView>(FindObjectsInactive.Include)?.transform.parent;
            Wire(village, gate, canvas, list);
        }

        public static void Wire(VillageController village, VillageGate gate, Transform canvas, DungeonDef[] list)
        {
            DungeonSelectView view = null;
            if (canvas != null)
            {
                view = canvas.GetComponentInChildren<DungeonSelectView>(true);
                if (view == null)
                {
                    var go = new GameObject("DungeonSelect", typeof(RectTransform));
                    go.transform.SetParent(canvas, false);
                    var rt = (RectTransform)go.transform;
                    rt.anchorMin = Vector2.zero; rt.anchorMax = Vector2.one;
                    rt.offsetMin = rt.offsetMax = Vector2.zero;
                    view = go.AddComponent<DungeonSelectView>();
                }
                // 화면 전환(아이리스)보다는 아래에 둔다
                var iris = canvas.Find("IrisTransition");
                if (iris != null) view.transform.SetSiblingIndex(iris.GetSiblingIndex());
            }

            var so = new SerializedObject(village);
            so.FindProperty("select").objectReferenceValue = view;
            so.FindProperty("gate").objectReferenceValue = gate;

            // 도리이 옆 행선지 석판 — 마을 NPC 목록에 끼워 넣는다 (대화·느낌표는 NPC와 같은 규칙)
            var worldProp = so.FindProperty("world").objectReferenceValue as GameObject;
            if (worldProp != null)
            {
                var stone = VillageBuilder.EnsureWaystone(worldProp.transform, gate != null ? gate.transform.parent : null);
                so.FindProperty("waystone").objectReferenceValue = stone.GetComponent<Waystone>();
                var npcs = so.FindProperty("npcs");
                bool has = false;
                for (int i = 0; i < npcs.arraySize; i++) if (npcs.GetArrayElementAtIndex(i).objectReferenceValue == stone) has = true;
                if (!has)
                {
                    npcs.arraySize++;
                    npcs.GetArrayElementAtIndex(npcs.arraySize - 1).objectReferenceValue = stone;
                }
            }
            var arr = so.FindProperty("dungeons");
            arr.arraySize = list.Length;
            for (int i = 0; i < list.Length; i++) arr.GetArrayElementAtIndex(i).objectReferenceValue = list[i];
            so.ApplyModifiedPropertiesWithoutUndo();
            EditorUtility.SetDirty(village);
            SetupBackdrops(village, list);
        }

        const string BackdropDir = "Assets/_Project/Textures/Backdrops/";

        /// <summary>
        /// 원경 배경판을 카메라에 붙이고, 마을·던전 테마에 그림을 넣는다.
        /// 그림은 Textures/Backdrops/Backdrop_&lt;Village|Forest|Mine|Snow|Rift&gt;.png
        /// </summary>
        public static void SetupBackdrops(VillageController village, DungeonDef[] list)
        {
            var cam = Camera.main != null ? Camera.main : Object.FindFirstObjectByType<Camera>();
            if (cam != null)
            {
                var bd = cam.GetComponentInChildren<Proto.Dungeon.Backdrop>(true);
                if (bd == null)
                {
                    var go = new GameObject("Backdrop");
                    go.transform.SetParent(cam.transform, false);
                    bd = go.AddComponent<Proto.Dungeon.Backdrop>();
                    var q = GameObject.CreatePrimitive(PrimitiveType.Quad);
                    q.name = "Image";
                    Object.DestroyImmediate(q.GetComponent<Collider>());
                    q.transform.SetParent(go.transform, false);
                    var r = q.GetComponent<Renderer>();
                    r.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off;
                    r.receiveShadows = false;
                    r.enabled = false;   // 그림이 들어오기 전엔 숨긴다 (편집 화면이 검게 덮이지 않게)
                    var bso = new SerializedObject(bd);
                    bso.FindProperty("quad").objectReferenceValue = r;
                    bso.ApplyModifiedPropertiesWithoutUndo();
                }
                var rend = bd.GetComponentInChildren<Renderer>(true);
                if (rend != null) rend.sharedMaterial = BackdropMaterial();
                // 원경판을 원거리 클립 안에 두기 위해
                if (cam.farClipPlane < 120f) cam.farClipPlane = 200f;
            }

            if (village != null)
            {
                var vso = new SerializedObject(village);
                vso.FindProperty("backdrop").objectReferenceValue = BackdropTex("Village");
                vso.ApplyModifiedPropertiesWithoutUndo();
            }
            var names = new Dictionary<string, string> { { "forest", "Forest" }, { "mine", "Mine" }, { "snow", "Snow" }, { "rift", "Rift" } };
            foreach (var d in list)
            {
                if (d == null || d.theme == null || !names.TryGetValue(d.id, out var n)) continue;
                var tex = BackdropTex(n);
                if (tex == null) continue;
                d.theme.backdrop = tex;
                EditorUtility.SetDirty(d.theme);
            }
            AssetDatabase.SaveAssets();
        }

        static Material BackdropMaterial()
        {
            const string path = "Assets/_Project/Materials/Backdrop.mat";
            var m = AssetDatabase.LoadAssetAtPath<Material>(path);
            var sh = Shader.Find("Proto/Backdrop");
            if (m == null) { m = new Material(sh); AssetDatabase.CreateAsset(m, path); }
            else if (m.shader != sh) m.shader = sh;
            EditorUtility.SetDirty(m);
            return m;
        }

        static Texture2D BackdropTex(string n)
        {
            string path = BackdropDir + "Backdrop_" + n + ".png";
            var imp = AssetImporter.GetAtPath(path) as TextureImporter;
            if (imp == null) return null;
            if (imp.mipmapEnabled || imp.wrapMode != TextureWrapMode.Clamp || imp.maxTextureSize < 2048)
            {
                imp.textureType = TextureImporterType.Default;
                imp.mipmapEnabled = false;
                imp.wrapMode = TextureWrapMode.Clamp;
                imp.maxTextureSize = 2048;
                imp.SaveAndReimport();
            }
            return AssetDatabase.LoadAssetAtPath<Texture2D>(path);
        }

        static DungeonDef Def(string file, string id, string name, string sub, string desc, StageTheme theme,
                              Color main, Color accent, int danger,
                              float eh, float ed, float bh, float bd, float reward)
        {
            string path = Dir + file + ".asset";
            var d = AssetDatabase.LoadAssetAtPath<DungeonDef>(path);
            if (d == null) { d = ScriptableObject.CreateInstance<DungeonDef>(); AssetDatabase.CreateAsset(d, path); }
            d.id = id; d.displayName = name; d.subtitle = sub; d.description = desc; d.theme = theme;
            d.portalColor = main; d.portalAccent = accent; d.danger = danger;
            d.enemyHealthMul = eh; d.enemyDamageMul = ed; d.bossHealthMul = bh; d.bossDamageMul = bd;
            d.rewardMul = reward;
            // 카드 그림 — Textures/Dungeons/Card_<파일이름>.png 가 있으면 쓴다
            string art = "Assets/_Project/Textures/Dungeons/Card_" + file + ".png";
            var imp = AssetImporter.GetAtPath(art) as TextureImporter;
            if (imp != null && (imp.mipmapEnabled || imp.maxTextureSize != 1024 || imp.textureType != TextureImporterType.Default))
            {
                imp.textureType = TextureImporterType.Default;
                imp.mipmapEnabled = false;
                imp.maxTextureSize = 1024;
                imp.wrapMode = TextureWrapMode.Clamp;
                imp.SaveAndReimport();
            }
            d.cardArt = AssetDatabase.LoadAssetAtPath<Texture2D>(art);
            EditorUtility.SetDirty(d);
            return d;
        }

        static StageTheme Clone(StageTheme src, string name)
        {
            string path = Data + name + ".asset";
            var t = AssetDatabase.LoadAssetAtPath<StageTheme>(path);
            if (t == null) { t = ScriptableObject.CreateInstance<StageTheme>(); AssetDatabase.CreateAsset(t, path); }
            EditorUtility.CopySerialized(src, t);
            t.name = name;
            return t;
        }

        static StageTheme Save(StageTheme t)
        {
            EditorUtility.SetDirty(t);
            Debug.Log($"[Proto] {t.stageName} 테마 — 경계 {Len(t.borderTrees)} / 옆 {Len(t.sideTrees)} / 채움 {Len(t.borderFill)} / " +
                      $"아래 {Len(t.lowBorder)} / 지형물 {Len(t.obstacles)} / 잡초 {Len(t.clutter)} / 몬스터 {Len(t.enemyModels)}·{Len(t.championModels)} / 보스 {(t.bossModel != null ? t.bossModel.name : "없음")}");
            return t;
        }

        static int Len(GameObject[] a) => a == null ? 0 : a.Length;

        /// <summary>숲 바닥 머티리얼을 복사해서 텍스처와 색만 바꾼다.</summary>
        /// <summary>
        /// 바닥 머티리얼. Textures/Ground/&lt;art&gt;.png (손으로 그린 반복 타일)가 있으면 그걸 쓰고,
        /// 없으면 팩의 지형 텍스처에 색을 입혀 쓴다. 그린 타일은 색이 이미 들어 있어 거의 흰색으로 둔다.
        /// </summary>
        public static Material Ground(string name, string art, string fallback, Color fallbackTint, Color artTint)
        {
            var custom = GroundArt(art);
            var m = Ground(name, custom != null ? AssetDatabase.GetAssetPath(custom) : fallback, custom != null ? artTint : fallbackTint);
            if (custom != null)
            {
                // 팩 노멀맵은 이 그림과 결이 맞지 않는다 — 그림 자체에 음영이 있다
                m.SetTexture("_BumpMap", null);
                m.DisableKeyword("_NORMALMAP");
                m.SetFloat("_Smoothness", 0.05f);
                EditorUtility.SetDirty(m);
            }
            return m;
        }

        /// <summary>반복 타일 임포트 — 반복, 밉맵(비스듬히 봐도 지글거리지 않게), 비등방 필터.</summary>
        public static Texture2D GroundArt(string art)
        {
            string path = "Assets/_Project/Textures/Ground/" + art + ".png";
            var imp = AssetImporter.GetAtPath(path) as TextureImporter;
            if (imp == null) return null;
            if (imp.wrapMode != TextureWrapMode.Repeat || !imp.mipmapEnabled || imp.anisoLevel < 4)
            {
                imp.textureType = TextureImporterType.Default;
                imp.wrapMode = TextureWrapMode.Repeat;
                imp.mipmapEnabled = true;
                imp.anisoLevel = 8;
                imp.maxTextureSize = 1024;
                imp.SaveAndReimport();
            }
            return AssetDatabase.LoadAssetAtPath<Texture2D>(path);
        }

        static Material Ground(string name, string texPath, Color tint)
        {
            string path = "Assets/_Project/Materials/" + name + ".mat";
            var src = AssetDatabase.LoadAssetAtPath<Material>("Assets/_Project/Materials/Forest Ground.mat");
            var m = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (m == null)
            {
                m = src != null ? new Material(src) : new Material(Shader.Find("Universal Render Pipeline/Lit"));
                AssetDatabase.CreateAsset(m, path);
            }
            var tex = AssetDatabase.LoadAssetAtPath<Texture2D>(texPath);
            if (tex != null) m.SetTexture("_BaseMap", tex);
            else Debug.LogWarning("[Proto] 바닥 텍스처 없음: " + texPath);
            m.SetColor("_BaseColor", tint);
            EditorUtility.SetDirty(m);
            return m;
        }

        static GameObject[] P(string folder, params string[] names)
        {
            var l = new List<GameObject>();
            var miss = new List<string>();
            foreach (var n in names)
            {
                var go = AssetDatabase.LoadAssetAtPath<GameObject>(folder + n + ".prefab");
                if (go != null) l.Add(go); else miss.Add(n);
            }
            if (miss.Count > 0) Debug.LogWarning($"[Proto] 프리팹 못 찾음 ({folder}): {string.Join(", ", miss)}");
            return l.ToArray();
        }

        /// <summary>폴더를 모르는 프리팹 — 이름이 정확히 같은 것을 프로젝트 전체에서 찾는다. 없으면 조용히 넘어간다.</summary>
        static GameObject[] Find(params string[] names)
        {
            var l = new List<GameObject>();
            foreach (var n in names)
                foreach (var g in AssetDatabase.FindAssets(n + " t:Prefab"))
                {
                    var p = AssetDatabase.GUIDToAssetPath(g);
                    if (System.IO.Path.GetFileNameWithoutExtension(p) != n) continue;
                    var go = AssetDatabase.LoadAssetAtPath<GameObject>(p);
                    if (go != null) { l.Add(go); break; }
                }
            return l.ToArray();
        }

        static GameObject[] A(params string[] names) => P(AnimalDir, names);

        static GameObject[] Join(params GameObject[][] parts)
        {
            var l = new List<GameObject>();
            foreach (var p in parts) if (p != null) l.AddRange(p);
            return l.ToArray();
        }
