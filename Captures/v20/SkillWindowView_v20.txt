        void Awake()
        {
            Build();
            _root.gameObject.SetActive(false);
        }

        void Start()
        {
            Book.Changed += () => { if (IsOpen) Refresh(); };
            RefreshHints();
        }

        void OnEnable() => GameInput.SchemeChanged += RefreshHints;
        void OnDisable() => GameInput.SchemeChanged -= RefreshHints;

        /// <summary>커맨드 표기 · 슬롯 키 · 안내 문구를 지금 장치에 맞춘다.</summary>
        void RefreshHints()
        {
            if (_cards.Count == 0) return;
            bool kb = GameInput.Scheme == InputScheme.Keyboard;
            // 왼쪽 위에서 오른쪽 아래로 물결처럼 뒤집힌다
            foreach (var c in _cards)
                InputFx.Set(c.Cmd, InputGlyphs.Command(SkillBook.All[c.Index].Cmd), (c.Index % 4 + c.Index / 4) * 0.035f);
            for (int i = 0; i < _slots.Count; i++) InputFx.Set(_slots[i].Num, InputGlyphs.Slot(i), 0.2f + i * 0.04f);
            InputFx.Set(_slotSub, kb ? "배운 스킬을 드래그하거나 A S D F로 등록" : $"스킬을 고르고 {InputGlyphs.Of(Act.Chord)} + 버튼으로 등록", 0.18f);
            InputFx.Set(_slotFoot, kb ? "드래그로 순서 변경  ·  우클릭으로 해제" : $"같은 칸을 한 번 더 누르면 해제  ·  {InputGlyphs.Of(Act.Interact)} 배우기/강화", 0.24f);
            InputFx.Set(_closeLabel, $"닫기 ({InputGlyphs.Of(Act.Cancel)})", 0.3f);
            if (IsOpen) ShowDetail();
        }

        public void Open()
        {
            _root.gameObject.SetActive(true);
            transform.SetAsLastSibling();
            Refresh();
            StartCoroutine(PopIn());
        }

        public void Close()
        {
            bool was = IsOpen;
            EndDrag();
            _root.gameObject.SetActive(false);
            if (was) Closed?.Invoke();
        }

        void Update()
        {
            if (!IsOpen) return;
            if (GameInput.CancelPressed) { Close(); return; }

            // 방향키·패드 — 카드 사이 이동 (4열)
            var d = GameInput.Navigate();
            if (d != Vector2Int.zero)
            {
                int n = SkillBook.All.Length, col = _selected % 4, row = _selected / 4;
                col = (col + d.x + 4) % 4;
                row = (row - d.y + n / 4) % (n / 4);
                Select(row * 4 + col);
            }
            if (GameInput.InteractPressed) DoAction();

            // A S D F / R1 + 버튼 — 고른 스킬을 그 칸에 (이미 있으면 해제)
            int slot = GameInput.SlotPressed();
            if (slot >= 0 && Book.Learned(_selected))
            {
                if (Book.Slot(slot) == _selected) Book.Clear(slot);
                else { Book.Assign(slot, _selected); StartCoroutine(Pop(_slots[slot].Root, 1.1f)); }
            }

            // 고른 카드는 은은하게 숨 쉰다
            float pulse = 0.22f + 0.12f * Mathf.Sin(Time.unscaledTime * 4f);
            foreach (var c in _cards)
            {
                var g = c.Glow.color;
                g.a = c.Index == _selected ? pulse : 0f;
                c.Glow.color = g;
            }
        }

        void Build()

            UiKit.Stretch(dim.rectTransform);

            border.type = Image.Type.Sliced;
            UiKit.Place(border.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(W + 4f, H + 4f));

            panel.type = Image.Type.Sliced;
            UiKit.Stretch(panel.rectTransform);
            panel.rectTransform.offsetMin = new Vector2(2f, 2f);
            panel.rectTransform.offsetMax = new Vector2(-2f, -2f);
            _panel = panel.rectTransform;
            // 금장 테두리가 있으면 기존 금색 테두리선은 숨긴다

            BuildHeader();
            BuildCards();
            BuildDetail();
            BuildSlots();

            _ghost = UiKit.Image(_root, "DragGhost", new Color(1f, 1f, 1f, 0.85f));
            _ghost.rectTransform.sizeDelta = new Vector2(84f, 84f);
            _ghost.raycastTarget = false;
            _ghost.gameObject.SetActive(false);
        }

        void BuildCards()
        {
            const float cw = 300f, ch = 140f, gap = 16f;
            float x0 = (W - (4 * cw + 3 * gap)) * 0.5f;
            for (int i = 0; i < SkillBook.All.Length; i++)
            {
                int col = i % 4, row = i / 4;
                _cards.Add(Card(i, new Vector2(x0 + col * (cw + gap), -136f - row * (ch + gap)), new Vector2(cw, ch)));
            }
        }

            var v = new CardView { Index = index };

            Corner(glow.rectTransform, new Vector2(0f, 1f), pos + new Vector2(-18f, 18f), size + new Vector2(36f, 36f));
            v.Glow = glow;

            v.Bg = UiKit.Image(_panel, "Card_" + d.Name, ColCard, UiArt.RoundRectFill, true);
            v.Bg.type = Image.Type.Sliced;
            v.Root = v.Bg.rectTransform;
            Corner(v.Root, new Vector2(0f, 1f), pos, size);
            v.Outline = UiKit.Image(v.Root, "Outline", ColEdge, UiArt.RoundRectOutline);
            v.Outline.type = Image.Type.Sliced;
            UiKit.Stretch(v.Outline.rectTransform);
            UiKit.FitOutline(v.Outline, SprCard, 3f, 20f);

            const float tx = 142f;

            nm.fontStyle = FontStyles.Bold;
            nm.enableAutoSizing = true; nm.fontSizeMax = 26f; nm.fontSizeMin = 18f;

            var cmd = UiKit.Text(v.Root, "Cmd", d.Cmd, 22, TextAlignmentOptions.Left);
            v.Cmd = cmd;
            Corner(cmd.rectTransform, new Vector2(0f, 1f), new Vector2(tx, -50f), new Vector2(size.x - tx - 10f, 28f));
            cmd.color = ColGold;

            Corner(role.rectTransform, new Vector2(0f, 1f), new Vector2(tx, -80f), new Vector2(size.x - tx - 10f, 24f));

            v.Status = UiKit.Text(v.Root, "Status", "", 18, TextAlignmentOptions.Left);
            Corner(v.Status.rectTransform, new Vector2(0f, 1f), new Vector2(tx, -106f), new Vector2(size.x - tx - 10f, 26f));

            h.BeginDrag = e => BeginDragSkill(index, -1, e);

            h.EndDrag = _ => EndDrag();

            h.Drop = () => DropOnSlot(index);
            var rest = v.Outline.color;
            h.Enter = () => { if (_dragSkill >= 0) v.Outline.color = ColGold; };
            h.Exit = () => v.Outline.color = rest;
            h.BeginDrag = e => { if (Book.Slot(index) >= 0) BeginDragSkill(Book.Slot(index), index, e); };
            h.Drag = MoveGhost;
            h.EndDrag = _ => EndDrag();
            return v;
        }

        // ───────────────────────────── 상태 ─────────────────────────────

        void Refresh()
        {
            RefreshHints();
            _sp.text = $"<color=#e8e4f0>보유 SP</color>   <size=44><color=#ffcc6b><b>{Book.Sp}</b></color></size>";
            Paint();

            for (int s = 0; s < _slots.Count; s++)
            {
                var v = _slots[s];
                int k = Book.Slot(s);
                bool has = k >= 0;
                v.Plus.SetActive(!has);
                v.Ring.gameObject.SetActive(has);
                v.X.SetActive(has);
                if (has) SetArt(v.Art, SkillBook.All[k].Icon, true);
            }

            ShowDetail();
        }

        /// <summary>카드 색만 다시 칠한다 (호버·선택이 바뀔 때).</summary>
        void Paint()
        {
            foreach (var c in _cards)
            {
                int i = c.Index;
                var d = SkillBook.All[i];
                bool learned = Book.Learned(i);
                bool sel = i == _selected;

                c.Bg.color = sel ? ColCardSel : ColCard;
                if (SprCard != null)
                {
                    c.Outline.sprite = sel && SprCardSel != null ? SprCardSel : SprCard;
                    c.Outline.color = sel ? Color.white : (i == _hover ? new Color(1f, 0.92f, 0.75f, 1f) : new Color(1f, 1f, 1f, 0.7f));
                }
                else c.Outline.color = sel ? ColGold : (i == _hover ? new Color(0.70f, 0.62f, 0.50f, 1f) : ColEdge);
                c.Ring.color = SprRing != null ? (learned ? Color.white : new Color(0.55f, 0.56f, 0.62f, 1f)) : (learned ? ColGold : ColSteel);
                c.Art.color = learned ? Color.white : new Color(0.62f, 0.60f, 0.68f, 1f);
                c.Lock.SetActive(!learned);

                if (learned)
                    c.Status.text = Book.Maxed(i)
                        ? $"<color=#ffcc6b>Lv. {Book.Level(i)} / {d.MaxLevel}  마스터</color>"
                        : $"<color=#9be37a>Lv. {Book.Level(i)} / {d.MaxLevel}</color>";
                else
                    c.Status.text = $"<color=#8f8aa3>미습득  ·  </color><color=#ffcc6b>{d.LearnCost} SP</color>";
            }
        }

        void ShowDetail()
        {
            int i = _selected;
            var d = SkillBook.All[i];
            bool learned = Book.Learned(i);
            int lv = Book.Level(i);

            _detRing.color = SprRing != null ? (learned ? Color.white : new Color(0.55f, 0.56f, 0.62f, 1f)) : (learned ? ColGold : ColSteel);
            SetArt(_detArt, d.Icon, learned);
            _detName.text = d.Name;

            bool ok; string reason; int cost;
            if (!learned)
            {
                _detLevel.text = "<color=#8f8aa3>미습득</color>  <color=#9be37a>→ Lv. 1</color>";
                _detHead.text = "배우면";
                _detBody.text = $"{d.Role}  ·  커맨드 <color=#ffcc6b>{d.Cmd}</color>";
                ok = Book.CanLearn(i, out reason);
                cost = d.LearnCost;
                _actLabel.text = $"배우기  ·  {cost} SP";
            }
            else if (Book.Maxed(i))
            {
                _detLevel.text = $"<color=#ffcc6b>Lv. {lv}  마스터</color>";
                _detHead.text = "마스터 효과";
                _detBody.text = d.Mastery;
                ok = false; reason = null; cost = 0;
                _actLabel.text = "최대 레벨";
            }
            else
            {
                _detLevel.text = $"<color=#9be37a>Lv. {lv}  →  {lv + 1}</color>";
                bool toMaster = lv + 1 >= d.MaxLevel;
                _detHead.text = toMaster ? "다음 레벨  <color=#ffcc6b>· 마스터</color>" : "다음 레벨";
                _detBody.text = toMaster ? $"{d.PerLevel}  ·  <color=#ffcc6b>{d.Mastery}</color>" : d.PerLevel;
                ok = Book.CanUpgrade(i, out reason);
                cost = d.CostPerLevel;
                _actLabel.text = $"강화  ·  {cost} SP";
            }

            // 막힌 이유가 SP가 아니면 (선행 조건) 그걸 보여 준다
            if (!ok && reason != null && reason != "SP가 부족합니다" && !learned)
                _detBody.text = $"<color=#ff8f80>{reason}</color>";
            else if (!ok && reason == "SP가 부족합니다")
                _detHead.text += "   <color=#ff8f80>SP가 부족합니다</color>";

            _act.interactable = ok;
            _actImg.color = ok ? Color.white : new Color(0.42f, 0.42f, 0.46f, 1f);
            _actLabel.color = ok ? new Color(0.22f, 0.13f, 0.05f, 1f) : new Color(0.75f, 0.73f, 0.80f, 1f);
        }

        void Select(int i)
        {
            if (_selected == i) return;
            _selected = i;
            Paint();
            ShowDetail();
            StartCoroutine(Pop(_cards[i].Root, 1.04f));
        }

        void DoAction()
        {
            int i = _selected;
            bool done = Book.Learned(i) ? Book.Upgrade(i) : Book.Learn(i);
            if (!done) return;
            StartCoroutine(Pop(_cards[i].Root, 1.08f));
            StartCoroutine(Pop(_actImg.rectTransform, 1.06f));
            if (Proto.Feel.Feel.I != null) Proto.Feel.Feel.I.KillSfx();
        }

        // ───────────────────────────── 끌기 ─────────────────────────────

        void BeginDragSkill(int skill, int fromSlot, PointerEventData e)
        {
            if (!Book.Learned(skill)) return;
            _dragSkill = skill;
            _dragFromSlot = fromSlot;
            _dropped = false;
            _ghost.sprite = Resources.Load<Sprite>("Icons/" + SkillBook.All[skill].Icon);
            _ghost.gameObject.SetActive(true);
            _ghost.transform.SetAsLastSibling();
            MoveGhost(e);
        }

        void MoveGhost(PointerEventData e)
        {
            if (_dragSkill < 0) return;
            RectTransformUtility.ScreenPointToLocalPointInRectangle(_root, e.position, null, out var lp);
            _ghost.rectTransform.anchoredPosition = lp;
        }

        void DropOnSlot(int slot)
        {
            if (_dragSkill < 0) return;
            _dropped = true;
            if (_dragFromSlot >= 0) Book.Swap(_dragFromSlot, slot);
            else Book.Assign(slot, _dragSkill);
            StartCoroutine(Pop(_slots[slot].Root, 1.1f));
        }

        void EndDrag()
        {
            // 퀵슬롯에서 끌어내 아무 칸에도 안 놓으면 해제
            if (_dragSkill >= 0 && _dragFromSlot >= 0 && !_dropped) Book.Clear(_dragFromSlot);
            _dragSkill = -1;
            _dragFromSlot = -1;
            if (_ghost != null) _ghost.gameObject.SetActive(false);
            foreach (var s in _slots) s.Outline.color = ColEdge;
        }

        // ───────────────────────────── 도구 ─────────────────────────────

        void IconDisc(RectTransform parent, string icon, Vector2 pos, float size, out Image ring, out Image art)
        {
            ring = UiKit.Image(parent, "Ring", SprRing != null ? Color.white : ColGold, SprRing != null ? SprRing : UiArt.MetalDisc);
            Corner(ring.rectTransform, new Vector2(0f, 0.5f), pos, new Vector2(size, size));
            var inner = UiKit.Image(ring.transform, "Inner", new Color(0.08f, 0.06f, 0.12f, 1f), UiArt.Disc);
            UiKit.Place(inner.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(size * 0.86f, size * 0.86f));
            inner.gameObject.AddComponent<Mask>().showMaskGraphic = true;
            art = UiKit.Image(inner.transform, "Art", Color.white);
            UiKit.Place(art.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(size * 0.81f, size * 0.81f));
            if (icon != null) SetArt(art, icon, true);
        }

        GameObject LockBadge(Transform ring)
        {
            var lk = UiKit.Image(ring, "Lock", new Color(0.06f, 0.07f, 0.10f, 0.95f), UiArt.Disc);
            UiKit.Place(lk.rectTransform, new Vector2(0.5f, 0f), new Vector2(0f, 8f), new Vector2(30f, 30f));
            var lockSpr = UiKit.Ui("IconLock");
            var li = UiKit.Image(lk.transform, "Icon", new Color(0.85f, 0.86f, 0.92f, 1f), lockSpr != null ? lockSpr : UiArt.Lock);
            li.preserveAspect = true;
            UiKit.Place(li.rectTransform, new Vector2(0.5f, 0.5f), Vector2.zero, new Vector2(18f, 18f));
            return lk.gameObject;
        }

        static void SetArt(Image img, string icon, bool bright)
        {
            img.sprite = Resources.Load<Sprite>("Icons/" + icon);
            img.color = img.sprite == null ? Color.clear : (bright ? Color.white : new Color(0.62f, 0.60f, 0.68f, 1f));
        }

        static void Line(RectTransform p, float y)
        {
            if (Resources.Load<Sprite>("UI/DividerLine") != null) { UiKit.Divider(p, y - 1f); return; }
            var line = UiKit.Image(p, "Line", new Color(1f, 1f, 1f, 0.12f));
            line.rectTransform.anchorMin = new Vector2(0f, 1f); line.rectTransform.anchorMax = new Vector2(1f, 1f);
            line.rectTransform.pivot = new Vector2(0.5f, 1f);
            line.rectTransform.offsetMin = new Vector2(26f, y - 2f); line.rectTransform.offsetMax = new Vector2(-26f, y);
        }

        static void Corner(RectTransform rt, Vector2 anchor, Vector2 pos, Vector2 size)
        {
            rt.anchorMin = rt.anchorMax = anchor;
            rt.pivot = anchor;
            rt.anchoredPosition = pos;
            rt.sizeDelta = size;
        }

        IEnumerator PopIn()
        {
            var cg = _panel.parent.GetComponent<CanvasGroup>();
            if (cg == null) cg = _panel.parent.gameObject.AddComponent<CanvasGroup>();
            float t = 0f;
            while (t < 0.18f)
            {
                t += Time.unscaledDeltaTime;
                float k = Mathf.Clamp01(t / 0.18f);
                cg.alpha = k;
                _panel.parent.localScale = Vector3.one * Mathf.Lerp(0.96f, 1f, 1f - (1f - k) * (1f - k));
                yield return null;
            }
            cg.alpha = 1f;
            _panel.parent.localScale = Vector3.one;
        }

        IEnumerator Pop(RectTransform rt, float peak)
        {
            float t = 0f;
            while (t < 0.2f)
            {
                t += Time.unscaledDeltaTime;
                float k = Mathf.Clamp01(t / 0.2f);
                rt.localScale = Vector3.one * Mathf.Lerp(peak, 1f, k);
                yield return null;
            }
            rt.localScale = Vector3.one;
        }
    }
