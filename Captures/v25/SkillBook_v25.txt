using System;

namespace Proto.Core
{
    /// <summary>스킬 하나의 정의. 숫자는 스킬-목록.md 기준.</summary>
    public class SkillDef
    {
        public string Name, Cmd, Role, Icon;
        /// <summary>0 = 시작 스킬, 1~3 = 전투 단계, -1 = 채굴·탐험 갈래 (선행 조건 없음)</summary>
        public int Tier;
        public int LearnCost;
        public int MaxLevel = 5;
        public int CostPerLevel = 1;
        public string PerLevel, Mastery;
    }

    /// <summary>
    /// 스킬 포인트(SP)와 배운 스킬, 퀵슬롯.
    ///
    ///   배우기  SP를 내고 Lv1로 해금 — 배운 스킬은 장착 없이 전부 쓴다
    ///   강화    레벨당 1 SP (궁극기 2 SP), Lv5에서 마스터 효과
    ///   퀵슬롯  커맨드를 몰라도 쓸 수 있게 하는 편의 — 제한이 아니다
    ///
    /// 커맨드: 방향 + Z(스킬 키) 또는 방향 + X(공격 키). 내려찍기는 Z 단독 — 기본 스킬.
    ///
    /// 단계 조건: 2단은 1단 하나 이상, 3단(궁극기)은 2단 둘 이상을 배워야 한다.
    /// </summary>
    public class SkillBook
    {
        public const int SlotCount = 4;

        /// <summary>테스트용 시작 SP. 수련 노드가 생기면 0으로 내린다.</summary>
        public const int StartingSp = 6;

        public static readonly SkillDef[] All =
        {
            new SkillDef { Name = "돌진베기",   Cmd = "→ Z",   Role = "접근 · 이동",     Icon = "skill_dash_slash",     Tier = 0, PerLevel = "피해 +15% · 거리 +0.5m",   Mastery = "끝에서 충격파 (주변 ×1.0)" },
            new SkillDef { Name = "올려베기",   Cmd = "↑ X",   Role = "띄우기 · 콤보",   Icon = "skill_rising_slash",   Tier = 0, PerLevel = "피해 +15% · 체공 +0.1초",  Mastery = "띄운 적이 받는 피해 +20%" },
            new SkillDef { Name = "내려찍기",   Cmd = "Z",     Role = "기본 스킬 · 다운", Icon = "skill_ground_slam",    Tier = 0, PerLevel = "피해 +15% · 범위 +10%",    Mastery = "슈퍼아머에도 경직" },
            new SkillDef { Name = "회전베기",   Cmd = "↓↑ Z",  Role = "광역 · 탈출",     Icon = "skill_whirlwind",      Tier = 1, LearnCost = 2, PerLevel = "타격 +1회",            Mastery = "주변 적을 끌어당긴다" },
            new SkillDef { Name = "연속 찌르기", Cmd = "→→ X",  Role = "경직 고정",       Icon = "skill_rapid_thrusts",  Tier = 1, LearnCost = 2, PerLevel = "타수 +1",              Mastery = "마지막 타 넉백 ×3" },
            new SkillDef { Name = "검기 날리기", Cmd = "←→ Z",  Role = "원거리 · 관통",   Icon = "skill_sword_wave",     Tier = 2, LearnCost = 4, PerLevel = "피해 +15% · 사거리 +1m", Mastery = "검기가 둘로 갈라진다" },
            new SkillDef { Name = "발도술",     Cmd = "↓→ Z",  Role = "보스 딜 · 단일",  Icon = "skill_iaido",          Tier = 2, LearnCost = 4, PerLevel = "피해 +15%",            Mastery = "누르고 있으면 기를 모은다 (최대 ×1.5)" },
            new SkillDef { Name = "반격 자세",   Cmd = "← X",   Role = "되받아치기",      Icon = "skill_counter_stance", Tier = 2, LearnCost = 4, PerLevel = "자세 시간 +0.05초",     Mastery = "반격 성공 시 재사용 초기화" },
            new SkillDef { Name = "전투 함성",   Cmd = "↑↓ Z",  Role = "버프",            Icon = "skill_war_cry",        Tier = 2, LearnCost = 4, PerLevel = "지속 +2초",            Mastery = "외칠 때 주변 적 경직" },
            new SkillDef { Name = "광맥 파쇄",   Cmd = "↓↓ X",  Role = "채굴",            Icon = "skill_ore_shatter",    Tier = -1, LearnCost = 3, PerLevel = "재사용 -2초 · 산출 감소 -10%p", Mastery = "산출량 감소 없음" },
            new SkillDef { Name = "질주",       Cmd = "←← X",  Role = "탐험 · 피로도",   Icon = "skill_sprint",         Tier = -1, LearnCost = 3, PerLevel = "지속 +0.5초",          Mastery = "질주 중 새 방 진입 비용 -50%" },
            new SkillDef { Name = "대지 가르기", Cmd = "↓→↑ Z", Role = "궁극기",          Icon = "skill_earth_splitter", Tier = 3, LearnCost = 8, CostPerLevel = 2, PerLevel = "피해 +20%", Mastery = "균열이 3초간 남아 지속 피해" },
        };

        readonly int[] _levels = new int[All.Length];
        readonly int[] _slots = new int[SlotCount];

        public int Sp { get; private set; }
        public event Action Changed;

        public SkillBook()
        {
            for (int i = 0; i < All.Length; i++) _levels[i] = All[i].Tier == 0 ? 1 : 0;
            for (int s = 0; s < SlotCount; s++) _slots[s] = -1;
            _slots[0] = 0; _slots[1] = 1;   // 시작 스킬 두 개는 미리 꽂아 둔다
            Sp = StartingSp;
        }

        public int Level(int i) => _levels[i];
        public bool Learned(int i) => _levels[i] > 0;
        public bool Maxed(int i) => _levels[i] >= All[i].MaxLevel;
        public int Slot(int s) => _slots[s];

        public void AddSp(int n) { Sp += n; Changed?.Invoke(); }

        /// <summary>배울 수 있나. 안 되면 이유를 돌려준다.</summary>
        public bool CanLearn(int i, out string reason)
        {
            var d = All[i];
            reason = null;
            if (Learned(i)) { reason = "이미 배움"; return false; }
            if (d.Tier == 2 && CountTier(1) < 1) { reason = "1단 스킬을 하나 이상 배워야 합니다"; return false; }
            if (d.Tier == 3 && CountTier(2) < 2) { reason = "2단 스킬을 둘 이상 배워야 합니다"; return false; }
            if (Sp < d.LearnCost) { reason = "SP가 부족합니다"; return false; }
            return true;
        }

        public bool CanUpgrade(int i, out string reason)
        {
            reason = null;
            if (!Learned(i)) { reason = "배우지 않은 스킬"; return false; }
            if (Maxed(i)) { reason = "마스터"; return false; }
            if (Sp < All[i].CostPerLevel) { reason = "SP가 부족합니다"; return false; }
            return true;
        }

        public bool Learn(int i)
        {
            if (!CanLearn(i, out _)) return false;
            Sp -= All[i].LearnCost;
            _levels[i] = 1;
            // 빈 퀵슬롯이 있으면 바로 꽂아 준다
            for (int s = 0; s < SlotCount; s++) if (_slots[s] < 0) { _slots[s] = i; break; }
            Changed?.Invoke();
            return true;
        }

        public bool Upgrade(int i)
        {
            if (!CanUpgrade(i, out _)) return false;
            Sp -= All[i].CostPerLevel;
            _levels[i]++;
            Changed?.Invoke();
            return true;
        }

        /// <summary>스킬을 칸에 넣는다. 다른 칸에 이미 있으면 그 칸과 맞바꾼다.</summary>
        public void Assign(int slot, int skill)
        {
            if (skill < 0 || !Learned(skill)) return;
            int prev = Array.IndexOf(_slots, skill);
            if (prev == slot) return;
            if (prev >= 0) _slots[prev] = _slots[slot];
            _slots[slot] = skill;
            Changed?.Invoke();
        }

        public void Swap(int a, int b)
        {
            if (a == b) return;
            (_slots[a], _slots[b]) = (_slots[b], _slots[a]);
            Changed?.Invoke();
        }

        public void Clear(int slot)
        {
            if (_slots[slot] < 0) return;
            _slots[slot] = -1;
            Changed?.Invoke();
        }

        // ── 저장 ──
        /// <summary>스킬은 이름으로 저장한다 — 목록 순서가 바뀌어도 맞는 스킬에 들어간다.</summary>
        public void Export(System.Collections.Generic.List<string> names, System.Collections.Generic.List<int> levels,
                           System.Collections.Generic.List<string> slots, out int sp)
        {
            for (int i = 0; i < All.Length; i++) if (_levels[i] > 0) { names.Add(All[i].Name); levels.Add(_levels[i]); }
            for (int s = 0; s < SlotCount; s++) slots.Add(_slots[s] >= 0 ? All[_slots[s]].Name : "");
            sp = Sp;
        }

        public void Import(System.Collections.Generic.List<string> names, System.Collections.Generic.List<int> levels,
                           System.Collections.Generic.List<string> slots, int sp)
        {
            for (int i = 0; i < All.Length; i++) _levels[i] = All[i].Tier == 0 ? 1 : 0;   // 시작 스킬은 언제나 배운 상태
            if (names != null && levels != null)
                for (int k = 0; k < names.Count && k < levels.Count; k++)
                {
                    int i = Array.FindIndex(All, d => d.Name == names[k]);
                    if (i >= 0) _levels[i] = Math.Max(_levels[i], Math.Min(levels[k], All[i].MaxLevel));
                }
            for (int s = 0; s < SlotCount; s++)
            {
                _slots[s] = -1;
                if (slots == null || s >= slots.Count || string.IsNullOrEmpty(slots[s])) continue;
                int i = Array.FindIndex(All, d => d.Name == slots[s]);
                if (i >= 0 && _levels[i] > 0 && Array.IndexOf(_slots, i) < 0) _slots[s] = i;
            }
            Sp = Math.Max(0, sp);
            Changed?.Invoke();
        }

        int CountTier(int t)
        {
            int n = 0;
            for (int i = 0; i < All.Length; i++) if (All[i].Tier == t && Learned(i)) n++;
            return n;
        }
    }
}
