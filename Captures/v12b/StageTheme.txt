using UnityEngine;

namespace Proto.Data
{
    /// <summary>
    /// 스테이지 하나의 겉모습. 방을 프리팹으로 미리 만들어 두지 않고
    /// 이 표를 보고 그때그때 꾸민다.
    ///
    /// 방 프리팹을 만들지 않는 이유:
    ///   맵이 매번 랜덤이라 출구 조합만 15가지다. 방 종류까지 곱하면
    ///   손으로 만들 수 있는 양이 아니고, 만들어도 같은 방이 반복돼 금방 들킨다.
    ///   적·광맥·출구도 이미 절차적으로 배치되므로 장식만 따로 손으로 두면
    ///   서로 겹치는 걸 막을 방법이 없다.
    ///
    /// 대신 배치에 규칙을 준다. 순수 랜덤 뿌리기는 노이즈로 보인다.
    ///   경계  — 방 밖을 나무로 둘러싼다. 출구만 뚫는다
    ///   지형물 — 방 안에 큰 것 몇 개. 시선을 끊어 방마다 인상을 다르게 만든다
    ///   잡초  — 바닥 잔디·꽃·버섯. 밀도만 맞으면 된다
    ///
    /// 스테이지 2, 3은 이 에셋을 하나 더 만들면 끝난다.
    /// </summary>
    [CreateAssetMenu(fileName = "StageTheme", menuName = "Proto/Stage Theme")]
    public class StageTheme : ScriptableObject
    {
        [Header("── 이름 ──")]
        public string stageName = "숲";

        [Header("── 바닥 ──")]
        [Tooltip("바닥 머티리얼. 단색 바닥은 잡초가 아무리 많아도 지저분해 보인다.\n" +
                 "타일링되는 텍스처가 깔려야 잡초가 '바닥 위에 난 것'으로 읽힌다.")]
        public Material groundMaterial;

        [Tooltip("바닥 1타일이 덮는 크기(m). 작을수록 촘촘하다.")]
        public float groundTileSize = 4f;

        [Header("── 색과 분위기 ──")]
        public Color groundColor = new Color(0.24f, 0.35f, 0.20f);
        public Color wallColor = new Color(0.10f, 0.14f, 0.10f);

        [Tooltip("카메라 배경색. 나무 사이로 보이는 '먼 곳'이다.\n" +
                 "검정으로 두면 뒷벽 위가 허공으로 뚫려 보인다. 안개색과 맞춰야 한다.")]
        public Color skyColor = new Color(0.70f, 0.80f, 0.74f);

        [Tooltip("안개. 방 경계 너머를 가려서 나무 벽이 끝나는 지점을 숨긴다.\n" +
                 "주의: 카메라가 18m 뒤에 있으므로 시작 거리는 그보다 훨씬 커야 한다.\n" +
                 "플레이 필드까지 24m, 뒷벽이 25~30m다. 25m 아래로 내리면 화면 전체가 뿌옇게 된다.")]
        public bool useFog = true;
        public Color fogColor = new Color(0.36f, 0.45f, 0.38f);
        public float fogStart = 14f;
        public float fogEnd = 42f;

        public Color ambientColor = new Color(0.42f, 0.47f, 0.40f);

        [Header("── 원경 ──")]
        [Tooltip("안개 너머에 보이는 먼 풍경 그림. 비워 두면 안개색 하늘만 보인다.")]
        public Texture2D backdrop;
        [Tooltip("그림을 이만큼 올린다 (그림 높이 비율). +면 그림 아래쪽(지평선)이 더 보인다")]
        [Range(0f, 1f)] public float backdropShift = 0.15f;
        [Tooltip("그림 위에 안개를 얼마나 덮을지. 클수록 장면 색에 녹아든다")]
        [Range(0f, 1f)] public float backdropHaze = 0.3f;
        [Tooltip("그림 벽을 세울 거리(m, 카메라 기준). 가까울수록 뒤쪽 나무 줄이 그림에 가려 그림이 더 많이 보인다.\n" +
                 "플레이 필드가 약 20m, 첫 줄 나무가 약 26m다.")]
        public float backdropDepth = 31f;
        public Color sunColor = new Color(1f, 0.96f, 0.85f);

        [Header("── 경계: 큰 나무 ──")]
        [Tooltip("방을 둘러싸는 나무. 여러 겹으로 심어 너머가 안 보이게 한다.")]
        public GameObject[] borderTrees;
        [Tooltip("겹 수. 1겹이면 사이로 뒤가 보인다.")]
        public int borderRows = 3;
        [Tooltip("한 겹 안에서 나무 사이 간격")]
        public float borderSpacing = 2.4f;
        [Tooltip("겹과 겹 사이 간격")]
        public float borderRowGap = 2.6f;
        public Vector2 borderScale = new Vector2(0.85f, 1.35f);
        [Tooltip("출구 좌우로 이만큼은 비워서 길을 낸다")]
        public float exitGapHalfWidth = 3.2f;

        [Header("── 동·서 변: 중간 크기 나무 ──")]
        [Tooltip("옆 변에는 거대 나무를 쓸 수 없다. 폭이 20m나 되어서 아무리 밀어내도\n" +
                 "벽에 붙은 플레이어를 덮는다. 뒷벽보다 한 단계 작은 나무를 쓴다.")]
        public GameObject[] sideTrees;
        public Vector2 sideTreeScale = new Vector2(0.8f, 1.2f);
        public float sideSpacing = 3.0f;

        [Header("── 경계: 발치 채움 ──")]
        [Tooltip("나무 밑동 사이를 메우는 덤불. 없으면 밑이 뻥 뚫려 보인다.")]
        public GameObject[] borderFill;
        public int borderFillCount = 40;
        public Vector2 borderFillScale = new Vector2(1.0f, 2.2f);

        [Header("── 카메라 쪽 변 (화면 아래) ──")]
        [Tooltip("벨트스크롤 시점에서는 카메라가 방의 남쪽 바깥에 있다.\n" +
                 "그쪽 변에 큰 나무를 심으면 캐릭터와 카메라 사이를 막아 플레이어가 안 보인다.")]
        public bool openCameraSide = true;

        [Tooltip("남쪽 변을 대신 채울 낮은 것. 비워 두면 아무것도 심지 않는다.\n" +
                 "방 경계는 읽히되 시야는 막지 않는 높이여야 한다.")]
        public GameObject[] lowBorder;
        public int lowBorderCount = 30;
        public Vector2 lowBorderScale = new Vector2(0.8f, 1.5f);

        [Tooltip("동·서 나무를 바깥으로 더 밀어내는 거리.\n" +
                 "큰 나무는 폭이 7m나 되어서 벽 바로 옆에 두면 벽에 붙은 플레이어를 덮는다.")]
        public float sideBorderOutset = 4f;

        [Header("── 몬스터 ──")]
        [Tooltip("이 스테이지에 나올 몬스터 모델. 스폰할 때 하나씩 뽑는다.\n" +
                 "프리팹을 종류별로 만들지 않고 여기서 갈아끼운다 — 2스테이지는 목록만 바꾸면 된다.")]
        public GameObject[] enemyModels;

        [Tooltip("챔피언 전용 모델. 비워 두면 일반과 같은 목록을 쓴다.\n" +
                 "덩치가 다른 모델을 넣으면 접두사 색과 함께 '다르다'가 더 확실해진다.")]
        public GameObject[] championModels;

        [Header("── 보스 ──")]
        [Tooltip("보스방에 나오는 보스 모델")]
        public GameObject bossModel;
        public float bossScale = 2.0f;
        public string bossName = "실버백";
        public string bossTitle = "숲의 주인";

        [Header("── 방 안: 지형물 ──")]
        [Tooltip("바위·그루터기·쓰러진 나무. 시선을 끊는 역할이라 크게.")]
        public GameObject[] obstacles;
        public Vector2Int obstacleCount = new Vector2Int(2, 5);
        public Vector2 obstacleScale = new Vector2(0.6f, 1.1f);
        [Tooltip("입장 지점에서 이만큼 안에는 놓지 않는다")]
        public float obstacleClearFromEntry = 4f;

        [Header("── 방 안: 잡초 ──")]
        [Tooltip("잔디·꽃·버섯·잔가지. 밀도만 맞으면 된다.")]
        public GameObject[] clutter;
        public Vector2Int clutterCount = new Vector2Int(45, 75);
        public Vector2 clutterScale = new Vector2(0.8f, 1.6f);

        [Tooltip("잡초 무리의 개수. 풀은 고르게 나지 않고 뭉쳐서 난다.\n" +
                 "균일하게 흩뿌리면 밀도가 어떻든 노이즈로 보인다.")]
        public int clutterClumps = 12;

        [Tooltip("무리 하나의 반경")]
        public float clutterClumpRadius = 2.2f;

        [Tooltip("무리에 속하지 않고 홀로 나는 비율")]
        [Range(0f, 1f)] public float clutterStrayRatio = 0.2f;

        [Header("── 성능 ──")]
        [Tooltip("장식에 붙어 있는 콜라이더를 전부 떼어낸다.\n" +
                 "장식이 길을 막으면 적이 끼고 플레이어가 답답해진다. 벽은 아레나가 이미 막고 있다.")]
        public bool stripColliders = true;

        [Tooltip("방을 다 꾸민 뒤 정적 배칭으로 합친다. 드로우콜이 크게 준다.")]
        public bool staticBatch = true;
    }
}
